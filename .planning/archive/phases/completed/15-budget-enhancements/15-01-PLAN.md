```javascript
function checkVarianceAlerts(lines) {
  const alerts = [];
  const thresholds = { warning: 0.8, danger: 1.0, critical: 1.15 };

  for (const line of lines) {
    if (line.budgeted <= 0) continue;

    const usageRatio = (line.committed + line.pending) / line.budgeted;

    if (usageRatio >= thresholds.critical) {
      alerts.push({
        severity: 'critical',
        costCode: line.costCode,
        description: line.description,
        message: `${Math.round(usageRatio * 100)}% of budget used`,
        variance: line.variance,
        budgeted: line.budgeted,
        spent: line.committed + line.pending
      });
    } else if (usageRatio >= thresholds.danger) {
      alerts.push({
        severity: 'danger',
        costCode: line.costCode,
        description: line.description,
        message: 'At or over budget',
        variance: line.variance,
        budgeted: line.budgeted,
        spent: line.committed + line.pending
      });
    } else if (usageRatio >= thresholds.warning) {
      alerts.push({
        severity: 'warning',
        costCode: line.costCode,
        description: line.description,
        message: `${Math.round(usageRatio * 100)}% of budget used`,
        variance: line.variance,
        budgeted: line.budgeted,
        spent: line.committed + line.pending
      });
    }
  }

  renderAlerts(alerts);
}

function renderAlerts(alerts) {
  const container = document.getElementById('varianceAlerts');
  const list = document.getElementById('alertsList');

  if (alerts.length === 0) {
    container.style.display = 'none';
    return;
  }

  // Sort by severity (critical first)
  const severityOrder = { critical: 0, danger: 1, warning: 2 };
  alerts.sort((a, b) => severityOrder[a.severity] - severityOrder[b.severity]);

  list.innerHTML = alerts.map(alert => `
    <div class="alert-item alert-${alert.severity}">
      <div class="alert-badge">${alert.severity.toUpperCase()}</div>
      <div class="alert-content">
        <span class="alert-code">${alert.costCode}</span>
        <span class="alert-desc">${alert.description}</span>
        <span class="alert-message">${alert.message}</span>
      </div>
      <div class="alert-amounts">
        <span class="alert-spent">${formatCurrency(alert.spent)}</span>
        <span class="alert-budget">of ${formatCurrency(alert.budgeted)}</span>
      </div>
    </div>
  `).join('');

  container.style.display = 'block';
}
```

```javascript
let categoryChart = null;

function renderCategoryChart(lines) {
  // Group by category
  const categories = {};
  for (const line of lines) {
    const cat = line.category || 'Other';
    if (!categories[cat]) {
      categories[cat] = { budgeted: 0, committed: 0, billed: 0 };
    }
    categories[cat].budgeted += line.budgeted || 0;
    categories[cat].committed += line.committed || 0;
    categories[cat].billed += line.billed || 0;
  }

  const labels = Object.keys(categories);
  const budgetData = labels.map(c => categories[c].budgeted);
  const committedData = labels.map(c => categories[c].committed);
  const billedData = labels.map(c => categories[c].billed);

  const ctx = document.getElementById('categoryChart').getContext('2d');

  if (categoryChart) {
    categoryChart.destroy();
  }

  categoryChart = new Chart(ctx, {
    type: 'bar',
    data: {
      labels: labels,
      datasets: [
        {
          label: 'Budget',
          data: budgetData,
          backgroundColor: 'rgba(88, 166, 255, 0.7)',
          borderColor: 'rgba(88, 166, 255, 1)',
```

### Task 4: Implement spend trend chart with forecasting
**File:** `public/budgets.html`

Add trend chart showing cumulative spend with forecast line:

```javascript
async function renderTrendChart(jobId, totals) {
  // Fetch invoice history for trend data
  const res = await fetch(`/api/invoices?job_id=${jobId}&status=approved,in_draw,paid`);
  const invoices = await res.json();

  // Group invoices by week
  const weeklySpend = {};
  const now = new Date();

  for (const inv of invoices) {
    const date = new Date(inv.approved_at || inv.created_at);
    const weekStart = getWeekStart(date);
    const key = weekStart.toISOString().split('T')[0];

    if (!weeklySpend[key]) {
      weeklySpend[key] = 0;
    }
    weeklySpend[key] += inv.amount || 0;
  }

  // Sort weeks and calculate cumulative
  const sortedWeeks = Object.keys(weeklySpend).sort();
  let cumulative = 0;
  const cumulativeData = sortedWeeks.map(week => {
    cumulative += weeklySpend[week];
    return { x: week, y: cumulative };
  });

  // Calculate forecast (simple linear projection)
  const totalBudget = totals.budgeted || 0;
  const currentSpend = totals.billed || 0;
  const remainingBudget = totalBudget - currentSpend;

  // Calculate average weekly spend
  const weekCount = sortedWeeks.length || 1;
  const avgWeeklySpend = currentSpend / weekCount;

  // Project future weeks
  const forecastData = [...cumulativeData];
  if (avgWeeklySpend > 0 && remainingBudget > 0) {
    const weeksToComplete = Math.ceil(remainingBudget / avgWeeklySpend);
    let forecastCumulative = currentSpend;

    for (let i = 1; i <= Math.min(weeksToComplete, 12); i++) {
      const futureDate = new Date(now);
      futureDate.setDate(futureDate.getDate() + (i * 7));
      const futureWeek = futureDate.toISOString().split('T')[0];
      forecastCumulative += avgWeeklySpend;
      forecastData.push({ x: futureWeek, y: Math.min(forecastCumulative, totalBudget) });
    }
  }

  const ctx = document.getElementById('trendChart').getContext('2d');

  if (trendChart) {
    trendChart.destroy();
  }

  trendChart = new Chart(ctx, {
    type: 'line',
    data: {
      datasets: [
        {
          label: 'Actual Spend',
          data: cumulativeData,
          borderColor: 'rgba(63, 185, 80, 1)',
          backgroundColor: 'rgba(63, 185, 80, 0.1)',
          fill: true,
          tension: 0.3
        },
        {
          label: 'Forecast',
          data: forecastData.slice(cumulativeData.length - 1),
          borderColor: 'rgba(210, 153, 34, 1)',
          borderDash: [5, 5],
          fill: false,
          tension: 0.3
        },
        {
          label: 'Budget',
          data: [
            { x: sortedWeeks[0] || now.toISOString().split('T')[0], y: totalBudget },
            { x: forecastData[forecastData.length - 1]?.x || now.toISOString().split('T')[0], y: totalBudget }
          ],
          borderColor: 'rgba(88, 166, 255, 0.7)',
          borderDash: [10, 5],
          fill: false,
          pointRadius: 0
        }
      ]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      plugins: {
        legend: {
          position: 'top',
          labels: { color: '#e6edf3' }
        },
        tooltip: {
          callbacks: {
            label: function(context) {
              return context.dataset.label + ': ' + formatCurrency(context.raw.y);
            }
          }
        }
      },
      scales: {
        x: {
          type: 'time',
          time: {
            unit: 'week',
            displayFormats: { week: 'MMM d' }
          },
          ticks: { color: '#8b949e' },
          grid: { color: 'rgba(48, 54, 61, 0.5)' }
        },
        y: {
          ticks: {
            color: '#8b949e',
            callback: function(value) {
              return '$' + (value / 1000).toFixed(0) + 'k';
            }
          },
          grid: { color: 'rgba(48, 54, 61, 0.5)' }
        }
      }
    }
  });
}

function getWeekStart(date) {
  const d = new Date(date);
  const day = d.getDay();
  const diff = d.getDate() - day;
  return new Date(d.setDate(diff));
}
```

```javascript
function setupChartToggles() {
  const toggles = document.querySelectorAll('.chart-toggle');
  toggles.forEach(toggle => {
    toggle.addEventListener('click', () => {
      // Update active toggle
      toggles.forEach(t => t.classList.remove('active'));
      toggle.classList.add('active');

      // Show corresponding chart panel
      const chartType = toggle.dataset.chart;
      document.querySelectorAll('.chart-panel').forEach(panel => {
        panel.classList.remove('active');
      });
      document.getElementById(`${chartType}ChartContainer`).classList.add('active');
    });
  });
}
```

```javascript
function updateForecastMetrics(totals, invoices) {
  // Calculate weekly burn rate
  const approvedInvoices = invoices.filter(i =>
    ['approved', 'in_draw', 'paid'].includes(i.status)
  );

  if (approvedInvoices.length === 0) {
    document.getElementById('burnRate').textContent = '$0/wk';
    document.getElementById('estCompletion').textContent = '--';
    return;
  }

  // Find date range
  const dates = approvedInvoices.map(i => new Date(i.approved_at || i.created_at));
  const minDate = new Date(Math.min(...dates));
  const maxDate = new Date(Math.max(...dates));
  const weekCount = Math.max(1, (maxDate - minDate) / (7 * 24 * 60 * 60 * 1000));

  const totalSpent = totals.billed || 0;
  const burnRate = totalSpent / weekCount;

  document.getElementById('burnRate').textContent = formatCurrency(burnRate) + '/wk';

  // Estimate completion
  const remaining = (totals.budgeted || 0) - totalSpent;
  if (burnRate > 0 && remaining > 0) {
    const weeksRemaining = Math.ceil(remaining / burnRate);
    const estDate = new Date();
    estDate.setDate(estDate.getDate() + (weeksRemaining * 7));
    document.getElementById('estCompletion').textContent =
      estDate.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
  } else if (remaining <= 0) {
    document.getElementById('estCompletion').textContent = 'Complete';
  } else {
    document.getElementById('estCompletion').textContent = '--';
  }
}
```
