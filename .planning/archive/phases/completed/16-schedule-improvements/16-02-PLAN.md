# Plan 16-02: Critical Path Highlighting

<objective>
Implement critical path calculation and visual highlighting in the Gantt chart. Users can see which tasks are on the critical path (zero slack - any delay impacts project end date). This completes SCH-03.
</objective>

## Context

**Existing state:**
- Tasks have `depends_on` UUID array for dependencies
- `planned_start`, `planned_end`, `planned_duration_days` stored per task
- Gantt bars colored by status (pending/in_progress/completed/blocked)
- No critical path calculation or visualization

**Gap:**
- No algorithm to calculate earliest/latest start and slack
- No visual distinction for critical path tasks
- Dependencies not visualized as connector lines

**Algorithm overview:**
1. Build task dependency graph
2. Forward pass: Calculate Earliest Start (ES) and Earliest Finish (EF)
3. Backward pass: Calculate Latest Start (LS) and Latest Finish (LF)
4. Slack = LS - ES (or LF - EF)
5. Critical path = all tasks with Slack = 0

## Tasks

### Task 1: Add critical path calculation function
**File:** `public/js/schedule.js`

```javascript
function calculateCriticalPath(tasks) {
  if (!tasks || tasks.length === 0) return { criticalTaskIds: new Set(), taskMetrics: {} };

  // Build lookup maps
  const taskMap = new Map();
  const dependents = new Map(); // taskId -> array of tasks that depend on it

  tasks.forEach(task => {
    taskMap.set(task.id, task);
    dependents.set(task.id, []);
  });

  // Build dependents map (reverse of depends_on)
  tasks.forEach(task => {
    const deps = task.depends_on || [];
    deps.forEach(depId => {
      if (dependents.has(depId)) {
        dependents.get(depId).push(task.id);
      }
    });
  });

  // Calculate duration in days for each task
  function getDuration(task) {
    if (task.planned_duration_days) return task.planned_duration_days;
    if (task.planned_start && task.planned_end) {
      const start = new Date(task.planned_start);
      const end = new Date(task.planned_end);
      return Math.max(1, Math.ceil((end - start) / (1000 * 60 * 60 * 24)) + 1);
    }
    return 1;
  }

  // Forward pass: Calculate ES (Earliest Start) and EF (Earliest Finish)
  const metrics = {};
  const projectStart = Math.min(...tasks.map(t =>
    t.planned_start ? new Date(t.planned_start).getTime() : Infinity
  ));

  function calculateES(taskId, visited = new Set()) {
    if (visited.has(taskId)) return 0; // Circular dependency protection
    if (metrics[taskId]?.es !== undefined) return metrics[taskId].es;

    visited.add(taskId);
    const task = taskMap.get(taskId);
    if (!task) return 0;

    const deps = task.depends_on || [];
    let es = 0;

    if (deps.length === 0) {
      // No dependencies - can start at project start
      es = 0;
    } else {
      // ES = max(EF of all dependencies)
      deps.forEach(depId => {
        const depEF = calculateEF(depId, visited);
        es = Math.max(es, depEF);
      });
    }

    const duration = getDuration(task);
    const ef = es + duration;

    if (!metrics[taskId]) metrics[taskId] = {};
    metrics[taskId].es = es;
    metrics[taskId].ef = ef;
    metrics[taskId].duration = duration;

    return es;
  }

  function calculateEF(taskId, visited = new Set()) {
    calculateES(taskId, visited);
    return metrics[taskId]?.ef || 0;
  }

  // Calculate ES/EF for all tasks
  tasks.forEach(task => calculateES(task.id, new Set()));

  // Find project end (max EF)
  const projectEnd = Math.max(...Object.values(metrics).map(m => m.ef || 0));

  // Backward pass: Calculate LF (Latest Finish) and LS (Latest Start)
  function calculateLF(taskId, visited = new Set()) {
    if (visited.has(taskId)) return projectEnd;
    if (metrics[taskId]?.lf !== undefined) return metrics[taskId].lf;

    visited.add(taskId);
    const taskDependents = dependents.get(taskId) || [];
    let lf = projectEnd;

    if (taskDependents.length === 0) {
      // No dependents - can finish at project end
      lf = projectEnd;
    } else {
      // LF = min(LS of all dependents)
      taskDependents.forEach(depId => {
        const depLS = calculateLS(depId, visited);
        lf = Math.min(lf, depLS);
      });
    }

    const duration = metrics[taskId]?.duration || 1;
    const ls = lf - duration;

    metrics[taskId].lf = lf;
    metrics[taskId].ls = ls;
    metrics[taskId].slack = ls - metrics[taskId].es;

    return lf;
  }

  function calculateLS(taskId, visited = new Set()) {
    calculateLF(taskId, visited);
    return metrics[taskId]?.ls || 0;
  }

  // Calculate LF/LS for all tasks
  tasks.forEach(task => calculateLF(task.id, new Set()));

  // Identify critical path (tasks with slack = 0)
  const criticalTaskIds = new Set();
  Object.entries(metrics).forEach(([taskId, m]) => {
    if (Math.abs(m.slack) < 0.001) { // Float comparison tolerance
      criticalTaskIds.add(taskId);
    }
  });

  return { criticalTaskIds, taskMetrics: metrics };
}
```

### Task 2: Apply critical path styling to Gantt bars
**File:** `public/js/schedule.js`

Update `renderGantt()` to calculate and apply critical path:

```javascript
function renderGantt() {
  const ganttContent = document.getElementById('ganttContent');
  const ganttHeader = document.getElementById('ganttHeader');
  if (!ganttContent || !ganttHeader) return;

  const filteredTasks = getFilteredTasks();
  if (filteredTasks.length === 0) {
    ganttContent.innerHTML = '<div class="empty-state">No tasks to display</div>';
    ganttHeader.innerHTML = '';
    return;
  }

  // Calculate critical path
  const { criticalTaskIds, taskMetrics } = calculateCriticalPath(filteredTasks);
  state.criticalPath = criticalTaskIds;
  state.taskMetrics = taskMetrics;

  // Calculate date range
  const { minDate, maxDate, days } = calculateDateRange(filteredTasks);

  // Render header
  renderGanttHeader(ganttHeader, minDate, days);

  // Render rows
  ganttContent.innerHTML = '';
  filteredTasks.forEach(task => {
    const row = renderGanttRow(task, minDate, days, criticalTaskIds.has(task.id));
    ganttContent.appendChild(row);
  });

  initDragHandlers();
}

function renderGanttRow(task, minDate, days, isCritical = false) {
  const row = document.createElement('div');
  row.className = 'gantt-row';

  // Task label
  const label = document.createElement('div');
  label.className = 'gantt-label';
  label.textContent = task.name;
  if (isCritical) {
    const criticalBadge = document.createElement('span');
    criticalBadge.className = 'critical-badge';
    criticalBadge.textContent = 'Critical';
    label.appendChild(criticalBadge);
  }
  row.appendChild(label);

  // Task bar
  const barContainer = document.createElement('div');
  barContainer.className = 'gantt-bar-container';

  // Calculate position
  const startDate = new Date(task.planned_start);
  const endDate = new Date(task.planned_end);
  const dayWidth = 30; // pixels per day

  const startOffset = Math.floor((startDate - minDate) / (1000 * 60 * 60 * 24));
  const duration = Math.ceil((endDate - startDate) / (1000 * 60 * 60 * 24)) + 1;

  const leftPx = startOffset * dayWidth;
  const widthPx = duration * dayWidth;

  const bar = document.createElement('div');
  bar.className = `gantt-bar-${task.status || 'pending'}`;
  if (isCritical) {
    bar.classList.add('critical-path');
  }
  bar.dataset.taskId = task.id;
  bar.style.left = `${leftPx}px`;
  bar.style.width = `${Math.max(widthPx, 20)}px`;
  bar.title = `${task.name}\n${formatDate(task.planned_start)} - ${formatDate(task.planned_end)}${isCritical ? '\n⚠ Critical Path' : ''}`;

  // Progress fill
  if (task.completion_percent > 0) {
    const progress = document.createElement('div');
    progress.className = 'gantt-progress';
    progress.style.width = `${Math.min(100, task.completion_percent)}%`;
    bar.appendChild(progress);
  }

  barContainer.appendChild(bar);
  row.appendChild(barContainer);

  return row;
}
```

function renderListView() {
  const container = document.getElementById('taskListBody');
  if (!container) return;

  const filteredTasks = getFilteredTasks();

  // Calculate critical path (same algorithm)
  const { criticalTaskIds } = calculateCriticalPath(filteredTasks);

  container.innerHTML = filteredTasks.map(task => {
    const isCritical = criticalTaskIds.has(task.id);
    return `
      <tr class="task-row ${isCritical ? 'critical-path' : ''}" onclick="openTaskModal('${task.id}')">
        <td class="task-name">${escapeHtml(task.name)}</td>
        <td>${escapeHtml(task.trade || '-')}</td>
        <td>${escapeHtml(task.phase || '-')}</td>
        <td>${formatDate(task.planned_start)}</td>
        <td>${formatDate(task.planned_end)}</td>
        <td>${task.actual_start ? formatDate(task.actual_start) : '-'}</td>
        <td>${task.actual_end ? formatDate(task.actual_end) : '-'}</td>
        <td>
          <div class="progress-bar-small">
            <div class="progress-fill" style="width: ${task.completion_percent || 0}%"></div>
          </div>
          <span class="progress-text">${task.completion_percent || 0}%</span>
        </td>
        <td><span class="status-badge status-${task.status}">${formatStatus(task.status)}</span></td>
      </tr>
    `;
  }).join('');
}
