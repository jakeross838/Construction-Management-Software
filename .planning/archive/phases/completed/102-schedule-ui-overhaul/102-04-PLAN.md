```javascript
function startInlineEdit(element, taskId, field) {
  // Don't start if already editing
  if (element.querySelector('input')) return;

  const currentValue = element.textContent.trim();
  const originalHtml = element.innerHTML;

  element.innerHTML = `
    <input type="text"
           class="inline-edit-input"
           value="${escapeHtml(currentValue)}"
           onblur="finishInlineEdit(this, '${taskId}', '${field}', '${escapeHtml(currentValue)}')"
           onkeydown="handleInlineEditKey(event, this, '${taskId}', '${field}', '${escapeHtml(currentValue)}')"
    />
  `;

  const input = element.querySelector('input');
  input.focus();
  input.select();
}

function handleInlineEditKey(event, input, taskId, field, originalValue) {
  if (event.key === 'Enter') {
    input.blur();
  } else if (event.key === 'Escape') {
    input.value = originalValue;
    input.blur();
  }
}

async function finishInlineEdit(input, taskId, field, originalValue) {
  const newValue = input.value.trim();

  // Restore display
  const container = input.parentElement;
  container.textContent = newValue || originalValue;

  // If unchanged, skip API call
  if (newValue === originalValue || !newValue) {
    container.textContent = originalValue;
    return;
  }

  try {
    const res = await fetch(`/api/schedules/tasks/${taskId}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ [field]: newValue })
    });

    if (!res.ok) throw new Error('Failed to update');
    showToast('Task updated', 'success');

    // Update local state
    const task = state.tasks.find(t => t.id === taskId);
    if (task) task[field] = newValue;
  } catch (err) {
    container.textContent = originalValue;
    showToast('Failed to update task', 'error');
  }
}
```

```javascript
async function quickStatusChange(select, taskId) {
  const newStatus = select.value;
  const task = state.tasks.find(t => t.id === taskId);
  const oldStatus = task?.status;

  // Update select styling immediately
  select.className = `quick-status-select status-${newStatus}`;

  // Prepare updates
  const updates = { status: newStatus };

  // Auto-update percent_complete based on status
  if (newStatus === 'complete') {
    updates.percent_complete = 100;
    updates.actual_end = new Date().toISOString().split('T')[0];
    if (!task.actual_start) {
      updates.actual_start = new Date().toISOString().split('T')[0];
    }
  } else if (newStatus === 'in_progress' && oldStatus === 'not_started') {
    updates.actual_start = updates.actual_start || new Date().toISOString().split('T')[0];
    if (task.percent_complete === 0) {
      updates.percent_complete = 10; // Start with some progress
    }
  }

  try {
    const res = await fetch(`/api/schedules/tasks/${taskId}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(updates)
    });

    if (!res.ok) throw new Error('Failed to update status');

    // Update local state
    if (task) {
      Object.assign(task, updates);
    }

    showToast(`Status: ${formatStatus(newStatus)}`, 'success');
    updateStats();
  } catch (err) {
    // Revert
    select.value = oldStatus;
    select.className = `quick-status-select status-${oldStatus}`;
    showToast('Failed to update status', 'error');
  }
}
```

```javascript
function toggleTaskSelection(taskId, selected) {
  if (selected) {
    state.selectedTasks.add(taskId);
  } else {
    state.selectedTasks.delete(taskId);
  }
  updateBulkActionsBar();
  updateRowSelection(taskId, selected);
}

function toggleSelectAll(selected) {
  if (selected) {
    state.tasks.forEach(t => state.selectedTasks.add(t.id));
  } else {
    state.selectedTasks.clear();
  }
  renderTaskList();
  updateBulkActionsBar();
}

function clearSelection() {
  state.selectedTasks.clear();
  renderTaskList();
  updateBulkActionsBar();
}

function updateBulkActionsBar() {
  const bar = document.getElementById('bulkActionsBar');
  const count = state.selectedTasks.size;

  bar.style.display = count > 0 ? 'flex' : 'none';
  document.getElementById('selectedCount').textContent = count;
}

function updateRowSelection(taskId, selected) {
  const rows = document.querySelectorAll(`.task-row`);
  rows.forEach(row => {
    const checkbox = row.querySelector('.task-checkbox');
    if (checkbox) {
      const rowTaskId = row.querySelector('[data-task-id]')?.dataset?.taskId;
      if (rowTaskId === taskId) {
        row.classList.toggle('selected', selected);
      }
    }
  });
}
```

```javascript
async function bulkSetStatus(newStatus) {
  if (state.selectedTasks.size === 0) return;

  const taskIds = [...state.selectedTasks];
  const updates = { status: newStatus };

  if (newStatus === 'complete') {
    updates.percent_complete = 100;
    updates.actual_end = new Date().toISOString().split('T')[0];
  }

  try {
    // Update all selected tasks
    await Promise.all(taskIds.map(taskId =>
      fetch(`/api/schedules/tasks/${taskId}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updates)
      })
    ));

    showToast(`Updated ${taskIds.length} tasks`, 'success');
    clearSelection();
    await loadSchedule();
  } catch (err) {
    showToast('Failed to update some tasks', 'error');
  }
}

async function bulkDelete() {
  if (state.selectedTasks.size === 0) return;

  if (!confirm(`Delete ${state.selectedTasks.size} tasks? This cannot be undone.`)) {
    return;
  }

  const taskIds = [...state.selectedTasks];

  try {
    await Promise.all(taskIds.map(taskId =>
      fetch(`/api/schedules/tasks/${taskId}`, { method: 'DELETE' })
    ));

    showToast(`Deleted ${taskIds.length} tasks`, 'success');
    clearSelection();
    await loadSchedule();
  } catch (err) {
    showToast('Failed to delete some tasks', 'error');
  }
}
```
