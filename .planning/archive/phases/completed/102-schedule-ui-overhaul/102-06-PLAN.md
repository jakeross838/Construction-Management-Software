2. Add mobile detection utility:
```javascript
function isMobileDevice() {
  return window.matchMedia('(max-width: 767px)').matches ||
         window.matchMedia('(pointer: coarse)').matches;
}

// Debounce for scroll performance
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}
```

let touchStartY = 0;
let isPulling = false;

document.getElementById('listView')?.addEventListener('touchstart', (e) => {
  if (window.scrollY === 0) {
    touchStartY = e.touches[0].clientY;
    isPulling = true;
  }
});

document.getElementById('listView')?.addEventListener('touchmove', (e) => {
  if (!isPulling) return;

  const touchY = e.touches[0].clientY;
  const pullDistance = touchY - touchStartY;

  if (pullDistance > 80 && window.scrollY === 0) {
    // Visual feedback
    document.getElementById('listView').style.transform = `translateY(${Math.min(pullDistance * 0.3, 50)}px)`;
  }
});

document.getElementById('listView')?.addEventListener('touchend', async (e) => {
  if (!isPulling) return;

  const listView = document.getElementById('listView');
  listView.style.transform = '';

  const touchY = e.changedTouches[0].clientY;
  const pullDistance = touchY - touchStartY;

  if (pullDistance > 100 && window.scrollY === 0) {
    showToast('Refreshing...', 'info');
    await loadSchedule();
    showToast('Refreshed', 'success');
  }

  isPulling = false;
  touchStartY = 0;
});

// Make calendar scrollable on touch
function initMobileCalendar() {
  const calendarView = document.getElementById('calendarView');
  if (!calendarView) return;

  // Enable horizontal swipe for month navigation
  let touchStartX = 0;

  calendarView.addEventListener('touchstart', (e) => {
    touchStartX = e.touches[0].clientX;
  });

  calendarView.addEventListener('touchend', (e) => {
    const touchEndX = e.changedTouches[0].clientX;
    const diff = touchStartX - touchEndX;

    if (Math.abs(diff) > 50) { // Minimum swipe distance
      if (diff > 0) {
        calendarNextMonth();
      } else {
        calendarPrevMonth();
      }
    }
  });
}

// Call in DOMContentLoaded
