function setView(view) {
  state.currentView = view;

  // Save preference
  localStorage.setItem('schedule_view', view);

  // Update button states
  document.querySelectorAll('.view-btn').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.view === view);
  });

  // Hide all views
  document.getElementById('listView').style.display = 'none';
  document.getElementById('ganttView').style.display = 'none';
  document.getElementById('calendarView').style.display = 'none';
  document.getElementById('agendaView').style.display = 'none';

  // Show selected view
  switch (view) {
    case 'list':
      document.getElementById('listView').style.display = 'block';
      renderTaskList();
      break;
    case 'gantt':
      document.getElementById('ganttView').style.display = 'block';
      renderFrappeGantt();
      break;
    case 'calendar':
      document.getElementById('calendarView').style.display = 'block';
      renderCalendarView();
      break;
    case 'agenda':
      document.getElementById('agendaView').style.display = 'block';
      renderAgendaView();
      break;
  }
}

// Load saved preference on init
function loadViewPreference() {
  const saved = localStorage.getItem('schedule_view');
  if (saved && ['list', 'gantt', 'calendar', 'agenda'].includes(saved)) {
    state.currentView = saved;
  }
}

function renderCalendarView() {
  const year = state.calendarDate.getFullYear();
  const month = state.calendarDate.getMonth();

  // Update header
  const monthNames = ['January', 'February', 'March', 'April', 'May', 'June',
                      'July', 'August', 'September', 'October', 'November', 'December'];
  document.getElementById('calendarMonthYear').textContent = `${monthNames[month]} ${year}`;

  // Build calendar grid
  const grid = document.getElementById('calendarGrid');
  const firstDay = new Date(year, month, 1);
  const lastDay = new Date(year, month + 1, 0);
  const startDay = firstDay.getDay(); // 0=Sun

  // Day headers
  let html = '<div class="calendar-row calendar-header-row">';
  ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].forEach(day => {
    html += `<div class="calendar-day-header">${day}</div>`;
  });
  html += '</div>';

  // Build weeks
  let day = 1;
  const today = new Date();

  for (let week = 0; week < 6; week++) {
    html += '<div class="calendar-row">';

    for (let dow = 0; dow < 7; dow++) {
      if ((week === 0 && dow < startDay) || day > lastDay.getDate()) {
        html += '<div class="calendar-cell empty"></div>';
      } else {
        const cellDate = new Date(year, month, day);
        const dateStr = cellDate.toISOString().split('T')[0];
        const isToday = isSameDay(cellDate, today);
        const isWeekend = dow === 0 || dow === 6;

        // Find tasks on this day
        const dayTasks = state.tasks.filter(task => {
          const start = task.planned_start ? new Date(task.planned_start) : null;
          const end = task.planned_end ? new Date(task.planned_end) : null;
          if (!start) return false;
          return cellDate >= start && cellDate <= (end || start);
        });

        html += `
          <div class="calendar-cell ${isToday ? 'today' : ''} ${isWeekend ? 'weekend' : ''}"
               data-date="${dateStr}"
               onclick="calendarDayClick('${dateStr}')">
            <div class="calendar-date">${day}</div>
            <div class="calendar-tasks">
              ${dayTasks.slice(0, 3).map(task => `
                <div class="calendar-task status-${task.status}"
                     onclick="event.stopPropagation(); openTaskModal('${task.id}')"
                     title="${escapeHtml(task.name)}">
                  ${escapeHtml(truncate(task.name, 15))}
                </div>
              `).join('')}
              ${dayTasks.length > 3 ? `<div class="calendar-more">+${dayTasks.length - 3} more</div>` : ''}
            </div>
          </div>
        `;
        day++;
      }
    }

    html += '</div>';
    if (day > lastDay.getDate()) break;
  }

  grid.innerHTML = html;
}

function calendarPrevMonth() {
  state.calendarDate.setMonth(state.calendarDate.getMonth() - 1);
  renderCalendarView();
}

function calendarNextMonth() {
  state.calendarDate.setMonth(state.calendarDate.getMonth() + 1);
  renderCalendarView();
}

function calendarToday() {
  state.calendarDate = new Date();
  renderCalendarView();
}

function calendarDayClick(dateStr) {
  // Open task modal with date pre-filled for new task
  openTaskModal();
  document.getElementById('taskPlannedStart').value = dateStr;
  document.getElementById('taskPlannedEnd').value = dateStr;
}

function renderAgendaView() {
  // Populate vendor filter
  populateAgendaVendorFilter();

  // Get filtered tasks
  let tasks = [...state.tasks];

  // Filter by vendor
  if (state.agendaVendorId === 'unassigned') {
    tasks = tasks.filter(t => !t.assigned_vendor_id);
  } else if (state.agendaVendorId) {
    tasks = tasks.filter(t => t.assigned_vendor_id === state.agendaVendorId);
  }

  // Filter by date
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const weekEnd = new Date(today);
  weekEnd.setDate(weekEnd.getDate() + 7);

  if (state.agendaDateFilter === 'today') {
    tasks = tasks.filter(t => {
      const start = t.planned_start ? new Date(t.planned_start) : null;
      return start && isSameDay(start, today);
    });
  } else if (state.agendaDateFilter === 'week') {
    tasks = tasks.filter(t => {
      const start = t.planned_start ? new Date(t.planned_start) : null;
      return start && start >= today && start <= weekEnd;
    });
  }

  // Sort by date
  tasks.sort((a, b) => {
    const dateA = a.planned_start ? new Date(a.planned_start) : new Date('9999-12-31');
    const dateB = b.planned_start ? new Date(b.planned_start) : new Date('9999-12-31');
    return dateA - dateB;
  });

  // Group by date
  const grouped = {};
  tasks.forEach(task => {
    const dateKey = task.planned_start || 'unscheduled';
    if (!grouped[dateKey]) grouped[dateKey] = [];
    grouped[dateKey].push(task);
  });

  // Render
  const content = document.getElementById('agendaContent');

  if (tasks.length === 0) {
    content.innerHTML = `
      <div class="agenda-empty">
        <p>No tasks match your filters.</p>
        ${state.agendaVendorId ? '<p>Try selecting a different vendor or date range.</p>' : ''}
      </div>
    `;
    return;
  }

  let html = '';
  Object.keys(grouped).sort().forEach(dateKey => {
    const dateTasks = grouped[dateKey];
    const isToday = dateKey !== 'unscheduled' && isSameDay(new Date(dateKey), today);

    html += `
      <div class="agenda-date-group ${isToday ? 'today' : ''}">
        <div class="agenda-date-header">
          <span class="agenda-date">${dateKey === 'unscheduled' ? 'Unscheduled' : formatAgendaDate(dateKey)}</span>
          <span class="agenda-task-count">${dateTasks.length} task${dateTasks.length !== 1 ? 's' : ''}</span>
        </div>
        <div class="agenda-tasks">
          ${dateTasks.map(task => renderAgendaTask(task)).join('')}
        </div>
      </div>
    `;
  });

  content.innerHTML = html;
}

function renderAgendaTask(task) {
  const vendor = state.vendors.find(v => v.id === task.assigned_vendor_id);
  const vendorName = vendor?.name || 'Unassigned';
  const tradeName = trades.find(t => t.id === task.trade_id)?.name || '';

  return `
    <div class="agenda-task status-${task.status}" onclick="openTaskModal('${task.id}')">
      <div class="agenda-task-main">
        <span class="agenda-task-name">${escapeHtml(task.name)}</span>
        <span class="agenda-task-vendor">${escapeHtml(vendorName)}</span>
      </div>
      <div class="agenda-task-meta">
        ${tradeName ? `<span class="agenda-task-trade">${escapeHtml(tradeName)}</span>` : ''}
        <span class="agenda-task-progress">${task.percent_complete || 0}%</span>
        <select class="agenda-status-select status-${task.status}"
                onclick="event.stopPropagation();"
                onchange="quickStatusChange(this, '${task.id}')">
          <option value="not_started" ${task.status === 'not_started' ? 'selected' : ''}>Not Started</option>
          <option value="in_progress" ${task.status === 'in_progress' ? 'selected' : ''}>In Progress</option>
          <option value="complete" ${task.status === 'complete' ? 'selected' : ''}>Complete</option>
          <option value="blocked" ${task.status === 'blocked' ? 'selected' : ''}>Blocked</option>
        </select>
      </div>
    </div>
  `;
}

function populateAgendaVendorFilter() {
  const select = document.getElementById('agendaVendorFilter');
  const currentValue = select.value;

  select.innerHTML = `
    <option value="">All Vendors</option>
    <option value="unassigned">Unassigned</option>
    ${state.vendors.map(v => `<option value="${v.id}">${escapeHtml(v.name)}</option>`).join('')}
  `;

  select.value = currentValue || state.agendaVendorId;
}

function filterAgenda() {
  state.agendaVendorId = document.getElementById('agendaVendorFilter').value;
  renderAgendaView();
}

function agendaToday() {
  state.agendaDateFilter = 'today';
  renderAgendaView();
}

function agendaThisWeek() {
  state.agendaDateFilter = 'week';
  renderAgendaView();
}

function formatAgendaDate(dateStr) {
  const date = new Date(dateStr + 'T00:00:00');
  const today = new Date();
  const tomorrow = new Date(today);
  tomorrow.setDate(tomorrow.getDate() + 1);

  if (isSameDay(date, today)) return 'Today';
  if (isSameDay(date, tomorrow)) return 'Tomorrow';

  return date.toLocaleDateString('en-US', {
    weekday: 'short',
    month: 'short',
    day: 'numeric'
  });
}
