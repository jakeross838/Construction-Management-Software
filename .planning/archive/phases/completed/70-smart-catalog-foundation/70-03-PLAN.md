<objective>
Update the catalog UI to display and edit all estimation/scheduling fields: labor hours, install duration, crew size, lead time, waste factor, coverage rate, quality tier, permit requirements, rough-in info, and warranty terms.

Purpose: Staff can view and manage all estimation data needed for project planning directly in the catalog interface.

Output: Enhanced product detail modal with estimation info sections and updated product form with new fields.
</objective>

<context>
@public/catalog.html - Product catalog page with detail modal and product form modal
@public/js/catalog.js - Catalog JavaScript (loadProducts, renderProductDetail, saveProduct, etc.)
@public/css/catalog.css - Catalog-specific styles
@.planning/codebase/CONVENTIONS.md - UI patterns and naming

**Current product detail modal sections:**
- Pricing (price, unit)
- Details (model/SKU, room, category)
- Specifications (specs JSONB)
- Dimensions (dimensions JSONB)
- Description
- Tags

**Current product form sections:**
- Basic info (name, category, vendor, model, room)
- Pricing (price, unit, default qty, sq footage)
- Description
- Specifications (color, finish, style)
- Dimensions (width, height, depth)
- Tags
- Image URL / spec sheet URL

**New sections needed:**
- Estimation & Labor (labor_hours, install_duration_hours, crew_size, lead_time_days)
- Material Coverage (waste_factor_percent, coverage_rate, coverage_unit)
- Quality Tier (quality_tier select)
- Permit Requirements (requires_permit, permit_type, rough_in_required, rough_in_notes)
- Warranty (warranty_months, warranty_notes)
</context>

**2. Update catalog.js - Enhance renderProductDetail function:**

Add after existing specs/dimensions/description rendering:

```javascript
// Estimation & Labor
const estimationSection = document.getElementById('estimationSection');
const hasEstimation = p.labor_hours || p.install_duration_hours || p.crew_size || p.lead_time_days;
if (hasEstimation) {
  estimationSection.style.display = 'block';
  let estimationHtml = '';

  if (p.labor_hours) {
    estimationHtml += `
      <div class="detail-item">
        <span class="detail-label">Labor Hours</span>
        <span class="detail-value">${p.labor_hours} hrs</span>
      </div>
    `;
  }
  if (p.install_duration_hours) {
    estimationHtml += `
      <div class="detail-item">
        <span class="detail-label">Install Duration</span>
        <span class="detail-value">${p.install_duration_hours} hrs</span>
      </div>
    `;
  }
  if (p.crew_size && p.crew_size > 1) {
    estimationHtml += `
      <div class="detail-item">
        <span class="detail-label">Crew Size</span>
        <span class="detail-value">${p.crew_size} people</span>
      </div>
    `;
  }
  if (p.lead_time_days) {
    estimationHtml += `
      <div class="detail-item">
        <span class="detail-label">Lead Time</span>
        <span class="detail-value">${p.lead_time_days} days</span>
      </div>
    `;
  }

  document.getElementById('estimationDetails').innerHTML = estimationHtml;
} else {
  estimationSection.style.display = 'none';
}

// Material Coverage
const coverageSection = document.getElementById('coverageSection');
const hasCoverage = p.waste_factor_percent || p.coverage_rate;
if (hasCoverage) {
  coverageSection.style.display = 'block';
  let coverageHtml = '';

  if (p.waste_factor_percent) {
    coverageHtml += `
      <div class="detail-item">
        <span class="detail-label">Waste Factor</span>
        <span class="detail-value">${p.waste_factor_percent}%</span>
      </div>
    `;
  }
  if (p.coverage_rate) {
    const coverageUnit = p.coverage_unit || 'sqft';
    coverageHtml += `
      <div class="detail-item">
        <span class="detail-label">Coverage Rate</span>
        <span class="detail-value">${p.coverage_rate} ${coverageUnit} per unit</span>
      </div>
    `;
  }

  document.getElementById('coverageDetails').innerHTML = coverageHtml;
} else {
  coverageSection.style.display = 'none';
}

// Quality Tier
const qualitySection = document.getElementById('qualitySection');
if (p.quality_tier) {
  qualitySection.style.display = 'block';
  const tierLabels = {
    builder: { label: 'Builder Grade', class: 'tier-builder' },
    standard: { label: 'Standard', class: 'tier-standard' },
    premium: { label: 'Premium', class: 'tier-premium' }
  };
  const tier = tierLabels[p.quality_tier] || { label: p.quality_tier, class: '' };
  document.getElementById('qualityTierDisplay').innerHTML = `
    <span class="quality-tier-badge ${tier.class}">${tier.label}</span>
  `;
} else {
  qualitySection.style.display = 'none';
}

// Permit Requirements
const permitSection = document.getElementById('permitSection');
if (p.requires_permit || p.rough_in_required) {
  permitSection.style.display = 'block';
  let permitHtml = '';

  if (p.requires_permit) {
    permitHtml += `
      <div class="permit-badge permit-required">
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <path d="M14 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z"></path>
          <polyline points="14 2 14 8 20 8"></polyline>
          <line x1="12" y1="18" x2="12" y2="12"></line>
          <line x1="9" y1="15" x2="15" y2="15"></line>
        </svg>
        <span>Permit Required${p.permit_type ? ': ' + escapeHtml(p.permit_type) : ''}</span>
      </div>
    `;
  }
  if (p.rough_in_required) {
    permitHtml += `
      <div class="permit-badge rough-in-required">
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <circle cx="12" cy="12" r="10"></circle>
          <line x1="12" y1="8" x2="12" y2="12"></line>
          <line x1="12" y1="16" x2="12.01" y2="16"></line>
        </svg>
        <span>Rough-In Required</span>
      </div>
    `;
    if (p.rough_in_notes) {
      permitHtml += `<p class="permit-notes">${escapeHtml(p.rough_in_notes)}</p>`;
    }
  }

  document.getElementById('permitInfo').innerHTML = permitHtml;
} else {
  permitSection.style.display = 'none';
}

// Warranty
const warrantySection = document.getElementById('warrantySection');
if (p.warranty_months || p.warranty_notes) {
  warrantySection.style.display = 'block';
  let warrantyHtml = '';

  if (p.warranty_months) {
    const years = Math.floor(p.warranty_months / 12);
    const months = p.warranty_months % 12;
    let durationText = '';
    if (years > 0) durationText += `${years} year${years > 1 ? 's' : ''}`;
    if (months > 0) durationText += `${years > 0 ? ' ' : ''}${months} month${months > 1 ? 's' : ''}`;
    warrantyHtml += `
      <div class="warranty-duration">
        <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <path d="M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z"></path>
        </svg>
        <span>${durationText} warranty</span>
      </div>
    `;
  }
  if (p.warranty_notes) {
    warrantyHtml += `<p class="warranty-notes">${escapeHtml(p.warranty_notes)}</p>`;
  }

  document.getElementById('warrantyInfo').innerHTML = warrantyHtml;
} else {
  warrantySection.style.display = 'none';
}
```
