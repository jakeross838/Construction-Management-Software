# Phase 20-01: Streamlined Approval Workflow

## Objective

Reduce clicks and friction in the invoice approval workflow through batch selection, quick actions, and smart defaults. Currently invoices require opening a fullscreen modal for every action — this plan adds lightweight alternatives.

## Context

**Current State:**
- Invoice approval requires: click card → modal opens → click Approve → confirm dialog → done (4 clicks minimum)
- No batch selection in invoice list
- Backend bulk approve endpoint exists (`POST /api/invoices/bulk/approve`) but frontend doesn't use it
- All corrections require opening the full edit modal

**Target State:**
- High-confidence invoices: select → batch approve (2 clicks)
- Quick actions directly on invoice cards (no modal required)
- Smart routing based on confidence level

## Requirement

**INV-AI-03**: User can approve invoices with fewer clicks through streamlined workflow and batch actions

## Success Criteria

1. High-confidence invoices can be approved with fewer clicks
2. Batch approval available for multiple invoices
3. Quick actions for common corrections (swap job, change amount)

### Task 2: Implement Batch Approve Function

**File:** `public/js/app.js`

Add batch approve functionality that calls existing backend endpoint:

```javascript
async function batchApproveSelected() {
  const ids = Array.from(selectedInvoices);
  if (ids.length === 0) return;

  // Validate all can be approved
  const invoices = ids.map(id => window.invoicesCache?.find(i => i.id === id)).filter(Boolean);
  const approvable = invoices.filter(i =>
    i.status === 'ready_for_approval' || i.status === 'needs_approval'
  );

  if (approvable.length !== ids.length) {
    const skipped = ids.length - approvable.length;
    if (!confirm(`${skipped} invoice(s) cannot be approved (wrong status). Continue with ${approvable.length}?`)) {
      return;
    }
  }

  try {
    const response = await fetch('/api/invoices/bulk/approve', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        invoice_ids: approvable.map(i => i.id),
        performed_by: 'Jake Ross' // TODO: get from auth
      })
    });

    const result = await response.json();

    if (result.success) {
      window.toasts?.success(`Approved ${result.approved} invoice(s)`);
      if (result.failed > 0) {
        window.toasts?.warning(`${result.failed} failed - check console`);
        console.warn('Batch approve failures:', result.results.failed);
      }
      clearSelection();
      toggleSelectionMode();
      await loadInvoices();
    } else {
      throw new Error(result.error || 'Batch approve failed');
    }
  } catch (err) {
    window.toasts?.error('Batch approve failed: ' + err.message);
  }
}

function canBatchApprove(invoiceId) {
  const inv = window.invoicesCache?.find(i => i.id === invoiceId);
  if (!inv) return false;
  return inv.status === 'ready_for_approval' || inv.status === 'needs_approval';
}
```

Add quick action handlers:

```javascript
async function quickApprove(invoiceId, event) {
  event.stopPropagation();

  try {
    const response = await fetch(`/api/invoices/${invoiceId}/approve`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ performed_by: 'Jake Ross' })
    });

    if (!response.ok) throw new Error('Approval failed');

    window.toasts?.success('Invoice approved');
    await loadInvoices();
  } catch (err) {
    window.toasts?.error(err.message);
  }
}

async function quickAddToDraw(invoiceId, event) {
  event.stopPropagation();

  const inv = window.invoicesCache?.find(i => i.id === invoiceId);
  if (!inv?.job_id) {
    window.toasts?.error('Invoice has no job assigned');
    return;
  }

  try {
    // Get or create draft draw
    const drawRes = await fetch(`/api/jobs/${inv.job_id}/current-draw?create=true`);
    const draw = await drawRes.json();

    // Add invoice to draw
    const addRes = await fetch(`/api/draws/${draw.id}/add-invoices`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ invoice_ids: [invoiceId] })
    });

    if (!addRes.ok) throw new Error('Failed to add to draw');

    window.toasts?.success('Added to Draw #' + draw.draw_number);
    await loadInvoices();
  } catch (err) {
    window.toasts?.error(err.message);
  }
}
```

Add quick correction modal functions:

```javascript
showQuickJobSwap() {
  const currentJob = this.currentInvoice?.job?.name || 'None';

  // Build job options from cache
  const jobOptions = (window.jobsCache || []).map(j =>
    `<option value="${j.id}" ${j.id === this.currentInvoice?.job_id ? 'selected' : ''}>${j.name}</option>`
  ).join('');

  this.showConfirmDialog({
    title: 'Quick Job Swap',
    message: `
      <div class="form-group">
        <label>Current Job</label>
        <div style="padding: 8px; background: var(--bg-card); border-radius: 4px; margin-bottom: 12px;">
          ${currentJob}
        </div>
      </div>
      <div class="form-group">
        <label>Change to:</label>
        <select id="quickJobSelect" class="field-input">
          ${jobOptions}
        </select>
      </div>
    `,
    confirmText: 'Update Job',
    type: 'info',
    onConfirm: async () => {
      const newJobId = document.getElementById('quickJobSelect').value;
      if (newJobId && newJobId !== this.currentInvoice?.job_id) {
        await this.quickUpdateField('job_id', newJobId);
      }
    }
  });
},

showQuickAmountFix() {
  const currentAmount = this.currentInvoice?.amount || 0;

  this.showConfirmDialog({
    title: 'Quick Amount Fix',
    message: `
      <div class="form-group">
        <label>Current Amount</label>
        <div style="padding: 8px; background: var(--bg-card); border-radius: 4px; margin-bottom: 12px;">
          ${window.Validation?.formatCurrency(currentAmount)}
        </div>
      </div>
      <div class="form-group">
        <label>Correct Amount:</label>
        <div class="amount-input-group">
          <span class="amount-prefix">$</span>
          <input type="text" id="quickAmountInput" class="field-input amount-field"
            value="${this.formatAmountInput(currentAmount)}"
            oninput="this.value = this.value.replace(/[^0-9.,]/g, '')">
        </div>
      </div>
    `,
    confirmText: 'Update Amount',
    type: 'info',
    onConfirm: async () => {
      const newAmount = parseFloat(document.getElementById('quickAmountInput').value.replace(/,/g, ''));
      if (!isNaN(newAmount) && newAmount !== currentAmount) {
        await this.quickUpdateField('amount', newAmount);
      }
    }
  });
},

async quickUpdateField(field, value) {
  try {
    const response = await fetch(`/api/invoices/${this.currentInvoice.id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ [field]: value })
    });

    if (!response.ok) throw new Error('Update failed');

    window.toasts?.success('Updated successfully');

    // Refresh the modal data
    await this.refreshCurrentInvoice();
  } catch (err) {
    window.toasts?.error(err.message);
  }
},

async refreshCurrentInvoice() {
  if (!this.currentInvoice?.id) return;

  const response = await fetch(`/api/invoices/${this.currentInvoice.id}`);
  const invoice = await response.json();

  this.currentInvoice = invoice;
  this.renderModalContent();
}
```
