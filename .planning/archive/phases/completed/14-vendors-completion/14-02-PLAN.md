# Plan 14-02: Vendor Documents System

<objective>
Enhance vendor document management with document listing, version tracking, and expiration alerts. This fulfills VND-02.
</objective>

## Context

**Existing state:**
- Document upload exists: `POST /api/vendors/:id/documents` (coi, w9, license)
- Single URL fields per doc type: `coi_url`, `w9_url`, `license_url`
- Boolean flags: `coi_on_file`, `w9_on_file`
- Expiration dates: `gl_expiration`, `wc_expiration`, `license_expiration`
- `/api/vendors/expiring` returns vendors with expiring docs

**Gap:**
- No document history (overwrite replaces previous)
- No way to list all documents for a vendor
- No document metadata (uploaded_by, upload_date)
- Frontend could improve document display

## Tasks

### Task 1: Create vendor documents table migration
**File:** `database/migration-048-vendor-documents.sql`

```sql
-- Migration 048: Vendor Documents Table
-- Stores document history and metadata

CREATE TABLE IF NOT EXISTS v2_vendor_documents (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  vendor_id UUID REFERENCES v2_vendors(id) ON DELETE CASCADE,
  document_type TEXT NOT NULL, -- 'coi', 'w9', 'license', 'contract', 'other'
  file_name TEXT NOT NULL,
  file_url TEXT NOT NULL,
  file_size INTEGER,
  uploaded_by TEXT,
  notes TEXT,
  expiration_date DATE,
  is_current BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for quick vendor document lookups
CREATE INDEX IF NOT EXISTS idx_vendor_documents_vendor ON v2_vendor_documents(vendor_id, document_type);
CREATE INDEX IF NOT EXISTS idx_vendor_documents_current ON v2_vendor_documents(vendor_id) WHERE is_current = true;
CREATE INDEX IF NOT EXISTS idx_vendor_documents_expiring ON v2_vendor_documents(expiration_date) WHERE is_current = true;

COMMENT ON TABLE v2_vendor_documents IS 'Stores all vendor documents with history tracking';
COMMENT ON COLUMN v2_vendor_documents.document_type IS 'Type: coi (Certificate of Insurance), w9, license, contract, other';
COMMENT ON COLUMN v2_vendor_documents.is_current IS 'Latest version of this document type for the vendor';
```

### Task 2: Add document list endpoint
**File:** `server/routes/vendors.js`

```javascript
// Get all documents for a vendor
router.get('/:id/documents', asyncHandler(async (req, res) => {
  const { id } = req.params;
  const { include_history } = req.query;

  let query = supabase
    .from('v2_vendor_documents')
    .select('*')
    .eq('vendor_id', id)
    .order('created_at', { ascending: false });

  // By default only current docs
  if (include_history !== 'true') {
    query = query.eq('is_current', true);
  }

  const { data, error } = await query;

  if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });
  res.json(data || []);
}));
```

### Task 3: Update document upload to use new table
**File:** `server/index.js` (update existing `/api/vendors/:id/documents` endpoint)

Update the upload endpoint to:
1. Mark previous doc of same type as `is_current = false`
2. Insert into `v2_vendor_documents` table
3. Still update vendor record for backward compatibility

```javascript
// Upload vendor document (COI, W-9, License)
app.post('/api/vendors/:id/documents', asyncHandler(async (req, res) => {
  const { id } = req.params;
  const { document_type, file_data, file_name, expiration_date, notes } = req.body;

  if (!document_type || !file_data) {
    return res.status(400).json({ error: 'document_type and file_data are required' });
  }

  const validTypes = ['coi', 'w9', 'license', 'contract', 'other'];
  if (!validTypes.includes(document_type)) {
    return res.status(400).json({ error: `Invalid document_type. Must be: ${validTypes.join(', ')}` });
  }

  // Decode base64 file
  const buffer = Buffer.from(file_data.split(',')[1] || file_data, 'base64');
  const fileName = `vendor-docs/${id}/${document_type}_${Date.now()}.pdf`;

  // Upload to Supabase storage
  const { data: uploadData, error: uploadError } = await supabase.storage
    .from('documents')
    .upload(fileName, buffer, {
      contentType: 'application/pdf',
      upsert: true
    });

  if (uploadError) {
    console.error('Upload error:', uploadError);
    throw new Error('Failed to upload document');
  }

  // Get public URL
  const { data: urlData } = supabase.storage
    .from('documents')
    .getPublicUrl(fileName);

  const publicUrl = urlData.publicUrl;

  // Mark previous version as not current
  await supabase
    .from('v2_vendor_documents')
    .update({ is_current: false })
    .eq('vendor_id', id)
    .eq('document_type', document_type)
    .eq('is_current', true);

  // Insert new document record
  const { data: docRecord, error: docError } = await supabase
    .from('v2_vendor_documents')
    .insert({
      vendor_id: id,
      document_type,
      file_name: file_name || `${document_type}.pdf`,
      file_url: publicUrl,
      file_size: buffer.length,
      expiration_date: expiration_date || null,
      notes: notes || null,
      is_current: true
    })
    .select()
    .single();

  if (docError) {
    console.error('Document record error:', docError);
    // Continue anyway - file is uploaded
  }

  // Update vendor record with document URL (backward compatibility)
  const updateField = {
    coi: { coi_url: publicUrl, coi_on_file: true },
    w9: { w9_url: publicUrl, w9_on_file: true, w9_received_date: new Date().toISOString().split('T')[0] },
    license: { license_url: publicUrl }
  }[document_type] || {};

  if (Object.keys(updateField).length > 0) {
    await supabase
      .from('v2_vendors')
      .update(updateField)
      .eq('id', id);
  }

  res.json({ success: true, url: publicUrl, document: docRecord });
}));
```

// Add function to load and render documents
async function loadVendorDocuments(vendorId) {
  try {
    const res = await fetch(`/api/vendors/${vendorId}/documents`);
    const docs = await res.json();
    renderVendorDocuments(docs);
  } catch (err) {
    console.error('Failed to load documents:', err);
  }
}

function renderVendorDocuments(docs) {
  const container = document.getElementById('vendorDocuments');
  if (docs.length === 0) {
    container.innerHTML = '<div class="empty-state-small">No documents</div>';
    return;
  }

  container.innerHTML = docs.map(doc => `
    <div class="document-item ${getDocStatusClass(doc)}">
      <div class="doc-icon">${getDocIcon(doc.document_type)}</div>
      <div class="doc-info">
        <span class="doc-type">${formatDocType(doc.document_type)}</span>
        <span class="doc-date">Uploaded ${formatDate(doc.created_at)}</span>
        ${doc.expiration_date ? `<span class="doc-expiry ${isExpiringSoon(doc.expiration_date) ? 'warning' : ''}">Expires ${formatDate(doc.expiration_date)}</span>` : ''}
      </div>
      <a href="${doc.file_url}" target="_blank" class="btn btn-sm btn-link">View</a>
    </div>
  `).join('');
}

function formatDocType(type) {
  const labels = { coi: 'Certificate of Insurance', w9: 'W-9', license: 'License', contract: 'Contract', other: 'Other' };
  return labels[type] || type;
}

function getDocIcon(type) {
  const icons = { coi: '📋', w9: '📄', license: '🪪', contract: '📝', other: '📁' };
  return icons[type] || '📄';
}
