## Verification Script

```javascript
// Run in browser console on price-intelligence.html

async function verifyPriceIntelligence() {
  console.log('=== Price Intelligence Verification ===\n');

  // Tab 1: Price Database
  console.log('Tab 1: Price Database');
  const stats = await fetch('/api/price-intelligence/stats').then(r => r.json());
  console.log('  Stats:', stats.total_items, 'items,', stats.active_vendors, 'vendors');

  const matrix = await fetch('/api/price-intelligence/matrix?limit=5').then(r => r.json());
  console.log('  Matrix:', matrix.items?.length, 'items loaded');

  // Tab 2: Order Optimizer
  console.log('\nTab 2: Order Optimizer');
  const parsed = await fetch('/api/order-optimizer/parse-list', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ text: '10 2x4x8 studs' })
  }).then(r => r.json());
  console.log('  Parse:', parsed.items?.length, 'items parsed');

  // Tab 3: Savings
  console.log('\nTab 3: Savings Tracker');
  const savings = await fetch('/api/savings/summary').then(r => r.json());
  console.log('  YTD Savings: $' + (savings.ytd?.savings_amount || 0).toFixed(2));

  // Tab 4: Spend
  console.log('\nTab 4: Spend Analytics');
  const spend = await fetch('/api/spend/by-vendor').then(r => r.json());
  console.log('  Vendors:', spend.vendor_count, 'Total Spend: $' + (spend.total_spend || 0).toFixed(2));

  const targets = await fetch('/api/spend/negotiation-targets').then(r => r.json());
  console.log('  Negotiation Targets:', targets.targets?.length);

  console.log('\n=== Verification Complete ===');
}

verifyPriceIntelligence();
```
