## Objective

Create backend fix endpoints that automatically correct validation errors identified by Phase 49 validation endpoints.

## Discovery

Phase 49 created validation endpoints with structured errors including `fix_hint` fields:
- `ORPHANED_PO_ALLOCATION` - allocation references deleted PO
- `ORPHANED_LINE_ITEM_ALLOCATION` - allocation references deleted line item
- `ORPHANED_CO_ALLOCATION` - allocation references deleted CO
- `CO_TOTAL_MISMATCH` - PO change_order_total doesn't match sum of approved COs
- `PO_TOTAL_MISMATCH` - PO total_amount doesn't match original + COs
- `ALLOCATION_SUM_EXCEEDS_INVOICE` - allocations exceed invoice amount

## Tasks

### Task 1: Add Fix Orphaned Allocation Endpoint

Create POST `/api/invoices/:id/fix-allocation` endpoint:

```javascript
// In server/routes/invoices.js

/**
 * Fix orphaned or problematic allocations
 * POST /api/invoices/:id/fix-allocation
 * Body: { allocation_id, fix_action: 'remove' | 'reassign', reassign_to?: { po_id, line_item_id, co_id } }
 */
router.post('/:id/fix-allocation', asyncHandler(async (req, res) => {
  const { id: invoiceId } = req.params;
  const { allocation_id, fix_action, reassign_to, performed_by } = req.body;

  if (!allocation_id || !fix_action) {
    return res.status(400).json({ error: 'allocation_id and fix_action required' });
  }

  // Verify allocation belongs to this invoice
  const { data: allocation } = await supabase
    .from('v2_invoice_allocations')
    .select('*')
    .eq('id', allocation_id)
    .eq('invoice_id', invoiceId)
    .single();

  if (!allocation) {
    return res.status(404).json({ error: 'Allocation not found' });
  }

  if (fix_action === 'remove') {
    // Delete the orphaned allocation
    await supabase.from('v2_invoice_allocations').delete().eq('id', allocation_id);

    // Log activity
    await logActivity('invoice', invoiceId, 'allocation_removed', {
      allocation_id,
      reason: 'fix_orphaned',
      performed_by
    });

    return res.json({ success: true, action: 'removed', allocation_id });
  }

  if (fix_action === 'reassign' && reassign_to) {
    // Update allocation with new PO/line item/CO references
    const updateFields = {};
    if (reassign_to.po_id !== undefined) updateFields.po_id = reassign_to.po_id;
    if (reassign_to.line_item_id !== undefined) updateFields.po_line_item_id = reassign_to.line_item_id;
    if (reassign_to.co_id !== undefined) updateFields.change_order_id = reassign_to.co_id;

    await supabase
      .from('v2_invoice_allocations')
      .update(updateFields)
      .eq('id', allocation_id);

    await logActivity('invoice', invoiceId, 'allocation_reassigned', {
      allocation_id,
      reassign_to,
      performed_by
    });

    return res.json({ success: true, action: 'reassigned', allocation_id, reassign_to });
  }

  res.status(400).json({ error: 'Invalid fix_action or missing reassign_to' });
}));
```

### Task 2: Add Fix PO Totals Endpoint

Create POST `/api/purchase-orders/:id/fix-totals` endpoint:

```javascript
// In server/routes/purchase-orders.js

/**
 * Fix PO total mismatches by recalculating from COs
 * POST /api/purchase-orders/:id/fix-totals
 * Body: { fix_actions: ['co_total', 'po_total'], performed_by }
 */
router.post('/:id/fix-totals', asyncHandler(async (req, res) => {
  const { id } = req.params;
  const { fix_actions = ['co_total', 'po_total'], performed_by } = req.body;

  // Get PO with original amount
  const { data: po } = await supabase
    .from('v2_purchase_orders')
    .select('id, po_number, original_amount, change_order_total, total_amount')
    .eq('id', id)
    .single();

  if (!po) {
    return res.status(404).json({ error: 'PO not found' });
  }

  // Calculate correct CO total from approved COs
  const { data: approvedCOs } = await supabase
    .from('v2_change_orders')
    .select('amount_change')
    .eq('po_id', id)
    .eq('status', 'approved');

  const calculatedCOTotal = (approvedCOs || []).reduce((sum, co) => sum + (co.amount_change || 0), 0);

  const updates = {};
  const fixes = [];

  if (fix_actions.includes('co_total')) {
    if (Math.abs((po.change_order_total || 0) - calculatedCOTotal) > 0.01) {
      updates.change_order_total = calculatedCOTotal;
      fixes.push({
        field: 'change_order_total',
        old_value: po.change_order_total,
        new_value: calculatedCOTotal
      });
    }
  }

  if (fix_actions.includes('po_total')) {
    const expectedTotal = (po.original_amount || 0) + calculatedCOTotal;
    if (Math.abs((po.total_amount || 0) - expectedTotal) > 0.01) {
      updates.total_amount = expectedTotal;
      fixes.push({
        field: 'total_amount',
        old_value: po.total_amount,
        new_value: expectedTotal
      });
    }
  }

  if (Object.keys(updates).length > 0) {
    updates.updated_at = new Date().toISOString();
    await supabase.from('v2_purchase_orders').update(updates).eq('id', id);

    await supabase.from('v2_po_activity').insert({
      po_id: id,
      action: 'totals_fixed',
      details: { fixes, performed_by }
    });
  }

  // Re-validate to confirm fix worked
  // Call the validate-totals logic inline
  const revalidation = await validatePOTotals(id);

  res.json({
    success: true,
    po_id: id,
    po_number: po.po_number,
    fixes_applied: fixes,
    validation_after: revalidation
  });
}));
```

### Task 3: Add Batch Fix Endpoint for Job

Create POST `/api/jobs/:id/fix-validation-errors` endpoint for bulk fixes:

```javascript
// In server/routes/jobs.js

/**
 * Batch fix validation errors for a job
 * POST /api/jobs/:id/fix-validation-errors
 * Body: { error_type, fix_action, performed_by }
 */
router.post('/:id/fix-validation-errors', asyncHandler(async (req, res) => {
  const { id: jobId } = req.params;
  const { error_type, fix_action, performed_by } = req.body;

  const results = { fixed: 0, failed: 0, details: [] };

  if (error_type === 'ORPHANED_PO_ALLOCATION' && fix_action === 'remove') {
    // Get all invoices for job
    const { data: invoices } = await supabase
      .from('v2_invoices')
      .select('id')
      .eq('job_id', jobId)
      .is('deleted_at', null);

    // Get all valid PO IDs for job
    const { data: validPOs } = await supabase
      .from('v2_purchase_orders')
      .select('id')
      .eq('job_id', jobId)
      .is('deleted_at', null);

    const validPOIds = new Set((validPOs || []).map(p => p.id));

    for (const invoice of (invoices || [])) {
      // Find orphaned allocations
      const { data: allocations } = await supabase
        .from('v2_invoice_allocations')
        .select('id, po_id')
        .eq('invoice_id', invoice.id)
        .not('po_id', 'is', null);

      for (const alloc of (allocations || [])) {
        if (alloc.po_id && !validPOIds.has(alloc.po_id)) {
          // Orphaned - remove it
          await supabase.from('v2_invoice_allocations').delete().eq('id', alloc.id);
          results.fixed++;
          results.details.push({ invoice_id: invoice.id, allocation_id: alloc.id, action: 'removed' });
        }
      }
    }
  }

  if (error_type === 'PO_TOTAL_MISMATCH' && fix_action === 'recalculate') {
    // Get all POs for job
    const { data: pos } = await supabase
      .from('v2_purchase_orders')
      .select('id')
      .eq('job_id', jobId)
      .is('deleted_at', null);

    for (const po of (pos || [])) {
      try {
        // Recalculate each PO
        const { data: approvedCOs } = await supabase
          .from('v2_change_orders')
          .select('amount_change')
          .eq('po_id', po.id)
          .eq('status', 'approved');

        const coTotal = (approvedCOs || []).reduce((sum, co) => sum + (co.amount_change || 0), 0);

        const { data: poData } = await supabase
          .from('v2_purchase_orders')
          .select('original_amount')
          .eq('id', po.id)
          .single();

        const expectedTotal = (poData.original_amount || 0) + coTotal;

        await supabase.from('v2_purchase_orders').update({
          change_order_total: coTotal,
          total_amount: expectedTotal,
          updated_at: new Date().toISOString()
        }).eq('id', po.id);

        results.fixed++;
        results.details.push({ po_id: po.id, action: 'recalculated' });
      } catch (err) {
        results.failed++;
        results.details.push({ po_id: po.id, action: 'failed', error: err.message });
      }
    }
  }

  // Log batch activity
  await supabase.from('v2_job_activity').insert({
    job_id: jobId,
    action: 'batch_fix',
    details: { error_type, fix_action, results, performed_by }
  });

  res.json({
    success: true,
    job_id: jobId,
    error_type,
    fix_action,
    results
  });
}));
```
