### Task 2: Create bids.js

**File:** `public/js/bids.js`

Core structure:

```javascript
/**
 * Bids Management
 */

let bids = [];
let jobs = [];
let vendors = [];
let selectedBids = new Set();

// ============================================================
// INITIALIZATION
// ============================================================

document.addEventListener('DOMContentLoaded', async () => {
  setupEventListeners();

  try {
    await Promise.all([
      loadJobs().catch(err => console.error('Jobs failed:', err)),
      loadVendors().catch(err => console.error('Vendors failed:', err))
    ]);
  } catch (err) {
    showToast('Some data failed to load', 'error');
  }

  await loadBids();
  await loadStats();
});

// ============================================================
// DATA LOADING
// ============================================================

async function loadBids() {
  const params = new URLSearchParams();
  const jobId = document.getElementById('jobFilter').value;
  const vendorId = document.getElementById('vendorFilter').value;
  const status = document.getElementById('statusFilter').value;

  if (jobId) params.append('job_id', jobId);
  if (vendorId) params.append('vendor_id', vendorId);
  if (status) params.append('status', status);

  const response = await fetch(`/api/bids?${params}`);
  bids = await response.json();
  renderBidsTable();
}

async function loadStats() {
  const jobId = document.getElementById('jobFilter').value;
  const params = jobId ? `?job_id=${jobId}` : '';
  const response = await fetch(`/api/bids/stats${params}`);
  const stats = await response.json();
  renderStats(stats);
}

// ============================================================
// RENDERING
// ============================================================

function renderBidsTable() {
  const tbody = document.getElementById('bidsTableBody');

  if (!bids.length) {
    tbody.innerHTML = `
      <tr>
        <td colspan="8" class="empty-state">
          No bids found. Click "New Bid" to create one.
        </td>
      </tr>
    `;
    return;
  }

  tbody.innerHTML = bids.map(bid => `
    <tr data-id="${bid.id}">
      <td>
        <input type="checkbox"
               onchange="toggleBidSelection('${bid.id}')"
               ${selectedBids.has(bid.id) ? 'checked' : ''}>
      </td>
      <td>
        <a href="#" onclick="openBidDetail('${bid.id}'); return false;">
          ${escapeHtml(bid.title)}
        </a>
      </td>
      <td>${bid.job?.name || '—'}</td>
      <td>${bid.vendor?.name || '—'}</td>
      <td>${formatCurrency(bid.bid_amount)}</td>
      <td><span class="status-badge status-${bid.status}">${formatStatus(bid.status)}</span></td>
      <td>${formatDate(bid.received_date)}</td>
      <td class="actions-cell">
        <button class="btn-icon" onclick="openEditBidModal('${bid.id}')" title="Edit">✏️</button>
        <button class="btn-icon" onclick="deleteBid('${bid.id}')" title="Delete">🗑️</button>
      </td>
    </tr>
  `).join('');
}

function renderStats(stats) {
  document.getElementById('statsGrid').innerHTML = `
    <div class="stat-card">
      <div class="stat-value">${stats.total}</div>
      <div class="stat-label">Total Bids</div>
    </div>
    <div class="stat-card stat-warning">
      <div class="stat-value">${stats.received}</div>
      <div class="stat-label">Received</div>
    </div>
    <div class="stat-card stat-info">
      <div class="stat-value">${stats.under_review}</div>
      <div class="stat-label">Under Review</div>
    </div>
    <div class="stat-card stat-success">
      <div class="stat-value">${stats.accepted}</div>
      <div class="stat-label">Accepted</div>
    </div>
    <div class="stat-card">
      <div class="stat-value">${formatCurrency(stats.total_amount)}</div>
      <div class="stat-label">Total Value</div>
    </div>
  `;
}
```

**Key Functions:**
- `openCreateBidModal()` - New bid form
- `openEditBidModal(id)` - Edit existing bid
- `openBidDetail(id)` - Full detail view with tabs
- `saveBid()` - Create/update via API
- `deleteBid(id)` - Soft delete with confirmation
- `changeStatus(id, newStatus)` - Status transition
- `uploadDocument(bidId)` - Document upload handler
- `openCompareModal()` - Side-by-side comparison
- `convertToPO(bidId)` - Trigger PO creation

### Task 5: Document Upload UI

Drag-and-drop zone in Documents tab:

```javascript
function setupDocumentUpload(bidId) {
  const dropzone = document.getElementById('documentDropzone');

  dropzone.addEventListener('dragover', (e) => {
    e.preventDefault();
    dropzone.classList.add('dragover');
  });

  dropzone.addEventListener('dragleave', () => {
    dropzone.classList.remove('dragover');
  });

  dropzone.addEventListener('drop', async (e) => {
    e.preventDefault();
    dropzone.classList.remove('dragover');

    const files = e.dataTransfer.files;
    for (const file of files) {
      await uploadDocument(bidId, file);
    }
  });
}

async function uploadDocument(bidId, file) {
  const formData = new FormData();
  formData.append('document', file);
  formData.append('uploaded_by', 'Current User'); // TODO: real user

  const response = await fetch(`/api/bids/${bidId}/documents`, {
    method: 'POST',
    body: formData
  });

  if (response.ok) {
    showToast('Document uploaded', 'success');
    await refreshBidDetail(bidId);
  } else {
    showToast('Upload failed', 'error');
  }
}
```
