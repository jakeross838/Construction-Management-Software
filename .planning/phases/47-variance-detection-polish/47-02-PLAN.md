Test cases to cover:

1. **Text normalization tests:**
   - Handles empty/null input
   - Removes punctuation correctly
   - Lowercases text
   - Handles extra whitespace

2. **Similarity calculation tests:**
   - Exact match = 1.0
   - Partial word overlap
   - No overlap = 0
   - Short words (< 3 chars) ignored
   - Abbreviations (e.g., "Elec" vs "Electrical")

3. **Line item matching tests:**
   - Exact description match
   - Similar description (> 0.3 score)
   - Amount matching (within 10%)
   - No match found (below threshold)
   - Model number matching (e.g., "Kohler K-6489" vs "K-6489-0")

4. **Variance detection tests:**
   - Invoice under PO budget - no variance
   - Invoice over PO budget - high severity warning
   - Line item amount mismatch
   - Unmatched line item
   - Line item matched to CO
   - Line item matched to VPO
   - Approaching limit warning (90%+)

const { test, expect } = require('@playwright/test');

test.describe('Variance Detection - Similarity', () => {
  const calculateSimilarity = (str1, str2) => {
    const normalizeText = (text) => {
      if (!text) return '';
      return text.toLowerCase().replace(/[^\w\s]/g, ' ').replace(/\s+/g, ' ').trim();
    };
    const words1 = new Set(normalizeText(str1).split(' ').filter(w => w.length > 2));
    const words2 = new Set(normalizeText(str2).split(' ').filter(w => w.length > 2));
    if (words1.size === 0 || words2.size === 0) return 0;
    const intersection = [...words1].filter(w => words2.has(w)).length;
    const union = new Set([...words1, ...words2]).size;
    return intersection / union;
  };

  test('exact match returns 1.0', () => {
    expect(calculateSimilarity('Rough Carpentry', 'Rough Carpentry')).toBe(1);
  });

  test('partial overlap', () => {
    const sim = calculateSimilarity('Rough Carpentry Labor', 'Rough Carpentry Materials');
    expect(sim).toBeGreaterThan(0.4);
    expect(sim).toBeLessThan(1);
  });

  test('no overlap returns 0', () => {
    expect(calculateSimilarity('Plumbing Fixtures', 'Electrical Wiring')).toBe(0);
  });

  test('handles abbreviations', () => {
    // Current implementation won't match these - documenting expected behavior
    const sim = calculateSimilarity('Elec Panel', 'Electrical Panel');
    // This will be 0.5 (Panel matches, Elec doesn't match Electrical)
    expect(sim).toBeGreaterThanOrEqual(0);
  });
});

function normalizeText(text) {
  if (!text) return '';
  let normalized = text
    .toLowerCase()
    .replace(/[^\w\s]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();

  // Expand common abbreviations (optional enhancement)
  const abbreviations = {
    'elec': 'electrical',
    'plbg': 'plumbing',
    'mech': 'mechanical',
    'carp': 'carpentry',
    'insul': 'insulation',
    'conc': 'concrete',
    'struct': 'structural'
  };

  const words = normalized.split(' ');
  const expanded = words.map(w => abbreviations[w] || w);
  return expanded.join(' ');
}
