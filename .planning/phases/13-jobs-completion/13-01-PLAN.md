## Context

Currently the jobs.js routes only have:
- GET / (list jobs)
- GET /:id (get single job)
- GET /:id/purchase-orders, /budget, /draws, /stats
- PATCH /:id/specs (update specs only)

Missing:
- POST / (create job)
- PATCH /:id (update job basic fields)
- DELETE /:id (soft delete)
- Status workflow (active → completed, on_hold)
- Activity/audit logging for job changes

The v2_jobs table has: id, name, address, client_name, contract_amount, status, created_at
It needs: deleted_at, updated_at columns for soft delete and tracking.

No v2_job_activity table exists - needs to be created.

## Requirements Coverage

- **JOB-01**: Job CRUD - User can create, update, and soft-delete jobs via API
- **JOB-02**: Job status workflow - Jobs transition through active/completed/on_hold with audit trail

```sql
ALTER TABLE v2_jobs ADD COLUMN IF NOT EXISTS deleted_at TIMESTAMPTZ;

-- Create job activity/audit log table
CREATE TABLE IF NOT EXISTS v2_job_activity (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  job_id UUID REFERENCES v2_jobs(id) ON DELETE CASCADE,
  action TEXT NOT NULL,
  performed_by TEXT,
  field_changes JSONB,
  previous_status TEXT,
  new_status TEXT,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_v2_job_activity_job_id ON v2_job_activity(job_id);
CREATE INDEX IF NOT EXISTS idx_v2_job_activity_created_at ON v2_job_activity(created_at);
```

### Task 2: Add Job CRUD routes

**File:** `server/routes/jobs.js`

Add these routes (using existing asyncHandler/AppError patterns from Phase 12):

**POST /** - Create job
```javascript
router.post('/', validateRequest({
  body: { name: { required: true } }
}), asyncHandler(async (req, res) => {
  const { name, address, client_name, contract_amount, status } = req.body;

  const { data, error } = await supabase
    .from('v2_jobs')
    .insert({
      name,
      address,
      client_name,
      contract_amount: contract_amount || null,
      status: status || 'active'
    })
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  // Log activity
  await supabase.from('v2_job_activity').insert({
    job_id: data.id,
    action: 'created',
    performed_by: req.body.created_by || 'User',
    notes: `Job "${name}" created`
  });

  res.status(201).json(data);
}));
```

**PATCH /:id** - Update job (basic fields, not specs)
```javascript
router.patch('/:id', validateRequest({
  params: { id: { type: 'uuid' } }
}), asyncHandler(async (req, res) => {
  const { id } = req.params;
  const allowedFields = ['name', 'address', 'client_name', 'contract_amount', 'status'];

  // Get current job for activity log
  const { data: current } = await supabase
    .from('v2_jobs')
    .select('*')
    .eq('id', id)
    .is('deleted_at', null)
    .single();

  if (!current) throw notFoundError('job', id);

  // Build update object with only allowed fields
  const updates = {};
  const changes = {};
  for (const field of allowedFields) {
    if (req.body[field] !== undefined && req.body[field] !== current[field]) {
      updates[field] = req.body[field];
      changes[field] = { old: current[field], new: req.body[field] };
    }
  }

  if (Object.keys(updates).length === 0) {
    return res.json(current); // No changes
  }

  updates.updated_at = new Date().toISOString();

  const { data, error } = await supabase
    .from('v2_jobs')
    .update(updates)
    .eq('id', id)
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  // Log activity
  const action = updates.status ? 'status_changed' : 'updated';
  await supabase.from('v2_job_activity').insert({
    job_id: id,
    action,
    performed_by: req.body.updated_by || 'User',
    field_changes: changes,
    previous_status: updates.status ? current.status : null,
    new_status: updates.status || null
  });

  res.json(data);
}));
```

**DELETE /:id** - Soft delete job
```javascript
router.delete('/:id', validateRequest({
  params: { id: { type: 'uuid' } }
}), asyncHandler(async (req, res) => {
  const { id } = req.params;

  // Verify job exists
  const { data: current } = await supabase
    .from('v2_jobs')
    .select('id, name')
    .eq('id', id)
    .is('deleted_at', null)
    .single();

  if (!current) throw notFoundError('job', id);

  // Soft delete
  const { error } = await supabase
    .from('v2_jobs')
    .update({ deleted_at: new Date().toISOString() })
    .eq('id', id);

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  // Log activity
  await supabase.from('v2_job_activity').insert({
    job_id: id,
    action: 'deleted',
    performed_by: req.body.deleted_by || 'User',
    notes: `Job "${current.name}" archived`
  });

  res.json({ success: true, message: 'Job archived' });
}));
```
