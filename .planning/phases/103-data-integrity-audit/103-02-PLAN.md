```javascript
function verifyBudgetIntegrity(data) {
  const lineTotal = data.lines.reduce((sum, l) => sum + (l.budgeted || 0), 0);
  const apiTotal = data.totals.budgeted || 0;
  const variance = Math.abs(lineTotal - apiTotal);

  return {
    valid: variance < 0.01, // Allow for floating point
    lineTotal,
    apiTotal,
    variance
  };
}
```
