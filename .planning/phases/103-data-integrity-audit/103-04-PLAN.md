async function loadDashboardStats() {
  try {
    const res = await fetch('/api/dashboard/stats');
    const stats = await res.json();

    document.getElementById('pendingInvoices').textContent = stats.pendingInvoices ?? '-';
    document.getElementById('approvedAmount').textContent = formatMoney(stats.approvedAmount ?? 0);
    document.getElementById('openPOs').textContent = stats.openPOs ?? '-';
