```javascript
// Add to state object
state.hasBaseline = false;
state.showBaseline = false;
state.baselineData = null;

// Set baseline
async function setBaseline() {
  // Check if baseline exists
  const res = await fetch(`/api/schedules/${state.schedule.id}/baseline`);
  const data = await res.json();

  if (data.baseline_set_at) {
    document.getElementById('baselineWarning').style.display = 'block';
  } else {
    document.getElementById('baselineWarning').style.display = 'none';
  }

  openBaselineModal();
}

async function confirmSetBaseline() {
  try {
    const res = await fetch(`/api/schedules/${state.schedule.id}/set-baseline`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ set_by: 'User', force: true })
    });

    if (!res.ok) throw new Error('Failed to set baseline');

    showToast('Baseline set successfully', 'success');
    closeBaselineModal();
    await loadBaseline();
    updateBaselineUI();
  } catch (err) {
    showToast(err.message, 'error');
  }
}

async function loadBaseline() {
  try {
    const res = await fetch(`/api/schedules/${state.schedule.id}/baseline`);
    state.baselineData = await res.json();
    state.hasBaseline = !!state.baselineData.baseline_set_at;
  } catch (err) {
    state.hasBaseline = false;
  }
}

function toggleBaselineView() {
  state.showBaseline = !state.showBaseline;
  const btn = document.getElementById('toggleBaselineBtn');
  btn.textContent = state.showBaseline ? 'Hide Baseline' : 'Show Baseline';
  renderSchedule();
}
```

```javascript
  // Find baseline data for this task
  const baselineTask = state.baselineData?.tasks?.find(t => t.id === task.id);

  let varianceClass = '';
  let varianceText = '-';
```

5. Add template functions:
```javascript
async function loadTemplates() {
  try {
    const res = await fetch('/api/schedules/templates');
    state.templates = await res.json();
  } catch (err) {
    console.error('Failed to load templates:', err);
    state.templates = [];
  }
}

function openSaveTemplateModal() {
  document.getElementById('templateName').value = '';
  document.getElementById('templateDescription').value = '';
  const modal = document.getElementById('saveTemplateModal');
  modal.style.display = 'flex';
  modal.classList.add('show');
}

async function saveAsTemplate() {
  const name = document.getElementById('templateName').value.trim();
  if (!name) {
    showToast('Template name is required', 'error');
    return;
  }

  try {
    const res = await fetch('/api/schedules/templates/from-schedule', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        source_schedule_id: state.schedule.id,
        name,
        description: document.getElementById('templateDescription').value.trim(),
        project_type: document.getElementById('templateProjectType').value
      })
    });

    if (!res.ok) throw new Error('Failed to save template');

    showToast('Template saved successfully', 'success');
    closeSaveTemplateModal();
    await loadTemplates();
  } catch (err) {
    showToast(err.message, 'error');
  }
}

async function openApplyTemplateModal() {
  await loadTemplates();

  const select = document.getElementById('templateSelect');
  select.innerHTML = '<option value="">Choose a template...</option>';
  state.templates.forEach(t => {
    select.innerHTML += `<option value="${t.id}">${t.name} (${t.project_type})</option>`;
  });

  document.getElementById('scheduleStartDate').value = new Date().toISOString().split('T')[0];
  document.getElementById('templatePreview').style.display = 'none';

  const modal = document.getElementById('applyTemplateModal');
  modal.style.display = 'flex';
  modal.classList.add('show');
}

async function applyTemplate() {
  const templateId = document.getElementById('templateSelect').value;
  const startDate = document.getElementById('scheduleStartDate').value;

  if (!templateId || !startDate) {
    showToast('Please select template and start date', 'error');
    return;
  }

  try {
    const res = await fetch(`/api/schedules/templates/${templateId}/apply`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        job_id: state.currentJobId,
        start_date: startDate,
        created_by: 'User'
      })
    });

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.error || 'Failed to apply template');
    }

    showToast('Schedule created from template', 'success');
    closeApplyTemplateModal();
    await loadSchedule();
  } catch (err) {
    showToast(err.message, 'error');
  }
}
```
