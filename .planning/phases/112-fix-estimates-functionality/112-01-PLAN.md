# Phase 112-01: Fix Estimate Creation and Display

## Objective
Fix critical bugs AND make the estimates page a fully functional, easy-to-use estimating tool for custom homes. The goal is not just to fix bugs, but to ensure the entire estimate workflow is intuitive, complete, and practical for real-world custom home building.

## Problem Statement
User cannot create estimates or add line items. The page shows "No Job Selected" even when a job is selected in the sidebar and the URL contains `?job=...` parameter. Multiple UI issues prevent basic estimate workflow from functioning.

Beyond fixing bugs, the estimating tool must support the complete custom home workflow:
- Quick estimate creation for new projects
- Easy line item entry with cost codes
- Section organization (Site Work, Framing, Finishes, etc.)
- Assembly templates for common items (bathrooms, kitchens, etc.)
- Client-friendly proposals
- Budget comparison and tracking

## Success Criteria

### Critical Bug Fixes (Must Have)
- [ ] User can create an estimate from the empty state
- [ ] User can add line items to an estimate using the modal
- [ ] Line item modal opens properly with all fields functional
- [ ] Auto-calculation (qty × unit cost = amount) works correctly
- [ ] Page correctly loads estimate when job is selected in sidebar

### Full Functionality (Complete Workflow)

- [ ] User can generate client-facing proposals (PDF)

### User Experience (Easy to Use)
- [ ] Minimal clicks to accomplish common tasks

## Root Cause Analysis

Based on conversation summary, the issues are:

1. **Empty state showing when estimate exists**
   - API call `/api/estimates?job_id=...` may be returning 0 results even though estimate exists
   - Initialization order: loading state shows first, then API call, then content

2. **Create estimate modal issues**
   - Job dropdown may not be populating correctly
   - Modal visibility may have CSS issues (`.show` class pattern)

3. **Line item modal issues**
   - "Add Item" button was showing "coming soon" toast (now implemented but may have bugs)
   - Modal may not open due to missing `.show` class or `currentEstimate` not being set

4. **Tab switching not working**
   - CSS rules missing for `.tab-content.active` display

async function saveEstimate() {
  console.log('[saveEstimate] Saving estimate');

  const id = document.getElementById('estimateId').value;
  const jobId = document.getElementById('formJob').value;
  const title = document.getElementById('formTitle').value.trim();
  const status = document.getElementById('formStatus').value;

  if (!jobId) {
    showToast('Please select a job', 'error');
    return;
  }
  if (!title) {
    showToast('Please enter a title', 'error');
    return;
  }

  const data = {
    job_id: jobId,
    title: title,
    status: status || 'draft'
  };

  console.log('[saveEstimate] Data to save:', data);

  try {
    let response;
    if (id) {
      // Update existing
      response = await fetch(`/api/estimates/${id}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      });
    } else {
      // Create new
      response = await fetch('/api/estimates', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      });
    }

    if (!response.ok) {
      const err = await response.json();
      throw new Error(err.error || 'Failed to save estimate');
    }

    const savedEstimate = await response.json();
    console.log('[saveEstimate] Saved successfully:', savedEstimate.id);

    showToast('Estimate saved', 'success');
    closeCreateModal();

    // Load the created/updated estimate immediately
    currentJobId = savedEstimate.job_id;
    await loadEstimateForJob(currentJobId);

  } catch (err) {
    console.error('[saveEstimate] Error:', err);
    showToast(err.message || 'Failed to save estimate', 'error');
  }
}
```

function openAddLineModal(lineId = null, sectionId = null) {
  console.log('[openAddLineModal] Opening line item modal');
  console.log('[openAddLineModal] currentEstimate:', currentEstimate);
  console.log('[openAddLineModal] lineId:', lineId, 'sectionId:', sectionId);

  if (!currentEstimate) {

    console.error('[openAddLineModal] currentEstimate is null - cannot add line items');
    return;
  }

  // Reset form
  document.getElementById('lineDescription').value = '';
  document.getElementById('lineQuantity').value = '1';
  document.getElementById('lineUnit').value = 'ea';
  document.getElementById('lineUnitCost').value = '0';
  document.getElementById('lineAmount').value = '0';
  document.getElementById('lineNotes').value = '';
  document.getElementById('lineItemId').value = lineId || '';

  // Populate cost code dropdown
  const ccSelect = document.getElementById('lineCostCode');
  ccSelect.innerHTML = '<option value="">Select Cost Code...</option>';
  costCodes.forEach(cc => {
    const opt = document.createElement('option');
    opt.value = cc.id;
    opt.textContent = `${cc.code} - ${cc.name}`;
    ccSelect.appendChild(opt);
  });

  // Populate section dropdown
  const secSelect = document.getElementById('lineSection');
  secSelect.innerHTML = '<option value="">No Section</option>';
  if (currentEstimate.sections) {
    currentEstimate.sections.forEach(sec => {
      const opt = document.createElement('option');
      opt.value = sec.id;
      opt.textContent = sec.name;
      if (sec.id === sectionId) opt.selected = true;
      secSelect.appendChild(opt);
    });
  }

  // Setup auto-calculation
  const qtyInput = document.getElementById('lineQuantity');
  const costInput = document.getElementById('lineUnitCost');
  const amountInput = document.getElementById('lineAmount');

  const calculateAmount = () => {
    const qty = parseFloat(qtyInput.value) || 0;
    const cost = parseFloat(costInput.value) || 0;
    const total = qty * cost;
    amountInput.value = total.toFixed(2);
    console.log('[calculateAmount] qty:', qty, 'cost:', cost, 'total:', total);
  };

  // Remove old listeners to prevent duplicates
  qtyInput.removeEventListener('input', calculateAmount);
  costInput.removeEventListener('input', calculateAmount);

  // Add new listeners
  qtyInput.addEventListener('input', calculateAmount);
  costInput.addEventListener('input', calculateAmount);

  // Show modal
  const modal = document.getElementById('lineItemModal');
  modal.style.display = 'flex';
  setTimeout(() => {
    modal.classList.add('show');
    console.log('[openAddLineModal] Modal shown');

function switchTab(tabName) {
  console.log('[switchTab] Switching to:', tabName);

  // Hide all tab content
  document.querySelectorAll('.tab-content').forEach(content => {
    content.classList.remove('active');
  });

  // Deactivate all tab buttons
  document.querySelectorAll('.tab').forEach(btn => {
    btn.classList.remove('active');
  });

  // Show selected tab content
  const content = document.getElementById(`tab-${tabName}`);
  if (content) {
    content.classList.add('active');
    console.log('[switchTab] Activated content:', content.id);
  }

  // Activate selected tab button
  const btn = document.querySelector(`.tab[data-tab="${tabName}"]`);
  if (btn) {
    btn.classList.add('active');
  }
}

// Initialize tabs on page load
function initTabs() {
  document.querySelectorAll('.tab').forEach(btn => {
    btn.addEventListener('click', () => {
      const tabName = btn.getAttribute('data-tab');
      switchTab(tabName);
    });
  });

  // Activate first tab by default
  const firstTab = document.querySelector('.tab');
  if (firstTab) {
    switchTab(firstTab.getAttribute('data-tab'));
  }
}

// Call in DOMContentLoaded
initTabs();
```
