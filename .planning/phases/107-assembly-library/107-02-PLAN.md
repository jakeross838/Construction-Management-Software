```javascript
router.post('/:id/sections', asyncHandler(async (req, res) => {
  const { id } = req.params;
  const { name, description } = req.body;

  // Get next sort_order
  const { data: existing } = await supabase
    .from('v2_estimate_sections')
    .select('sort_order')
    .eq('estimate_id', id)
    .order('sort_order', { ascending: false })
    .limit(1);

  const nextOrder = (existing?.[0]?.sort_order || 0) + 1;

  const { data, error } = await supabase
    .from('v2_estimate_sections')
    .insert({
      estimate_id: id,
      name: name || 'New Section',
      description,
      sort_order: nextOrder
    })
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  res.status(201).json(data);
}));
```

3. **PATCH /api/estimates/:id/sections/:sectionId** - Update section:

```javascript
router.patch('/:id/sections/:sectionId', asyncHandler(async (req, res) => {
  const { id, sectionId } = req.params;
  const { name, description, sort_order } = req.body;

  const updates = {};
  if (name !== undefined) updates.name = name;
  if (description !== undefined) updates.description = description;
  if (sort_order !== undefined) updates.sort_order = sort_order;

  const { data, error } = await supabase
    .from('v2_estimate_sections')
    .update(updates)
    .eq('id', sectionId)
    .eq('estimate_id', id)
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  res.json(data);
}));
```

4. **DELETE /api/estimates/:id/sections/:sectionId** - Delete section:

```javascript
router.delete('/:id/sections/:sectionId', asyncHandler(async (req, res) => {
  const { id, sectionId } = req.params;

  // First, clear section_id from all lines in this section (keep lines, just unsection them)
  await supabase
    .from('v2_estimate_lines')
    .update({ section_id: null })
    .eq('section_id', sectionId);

  // Then delete the section
  const { error } = await supabase
    .from('v2_estimate_sections')
    .delete()
    .eq('id', sectionId)
    .eq('estimate_id', id);

  if (error) throw new AppError('DATABASE_ERROR', error.message);

  res.json({ success: true, message: 'Section deleted, items preserved' });
}));
```

5. **POST /api/estimates/:id/sections/reorder** - Reorder sections:
```javascript
router.post('/:id/sections/reorder', asyncHandler(async (req, res) => {
  const { id } = req.params;
  const { section_ids } = req.body;

  if (!Array.isArray(section_ids)) {
    throw new AppError('VALIDATION_ERROR', 'section_ids must be an array');
  }

  // Update sort_order for each section
  for (let i = 0; i < section_ids.length; i++) {
    await supabase
      .from('v2_estimate_sections')
      .update({ sort_order: i + 1 })
      .eq('id', section_ids[i])
      .eq('estimate_id', id);
  }

  res.json({ success: true });
}));
```

```javascript
async function loadEstimateDetail(estimateId) {
  const res = await fetch(`/api/estimates/${estimateId}`);
  currentEstimate = await res.json();

  // Process sections and lines
  currentEstimate.sections = currentEstimate.sections || [];
  currentEstimate.lines = currentEstimate.lines || [];

  // Group lines by section
  const linesBySection = {};
  const unsectionedLines = [];

  currentEstimate.lines.forEach(line => {
    if (line.section_id) {
      if (!linesBySection[line.section_id]) linesBySection[line.section_id] = [];
      linesBySection[line.section_id].push(line);
    } else {
      unsectionedLines.push(line);
    }
  });

  currentEstimate.linesBySection = linesBySection;
  currentEstimate.unsectionedLines = unsectionedLines;

  renderEstimateDetail();
}
```

```javascript
function renderSections() {
  const container = document.getElementById('sectionsContainer');
  if (!container || !currentEstimate) return;

  const sections = currentEstimate.sections || [];

  container.innerHTML = sections.map(section => {
    const sectionLines = currentEstimate.linesBySection[section.id] || [];
    const subtotal = sectionLines.reduce((sum, l) => sum + (l.parent_line_id ? 0 : Number(l.amount || 0)), 0);

    return `
      <div class="estimate-section" data-section-id="${section.id}" draggable="true">
        <div class="section-header">
          <span class="drag-handle" onmousedown="this.parentElement.parentElement.draggable=true"
                onmouseup="this.parentElement.parentElement.draggable=false">⋮⋮</span>
          <input type="text" class="section-name-input" value="${escapeHtml(section.name)}"
                 onchange="updateSectionName('${section.id}', this.value)"
                 onblur="updateSectionName('${section.id}', this.value)">
          <span class="section-subtotal">${formatCurrency(subtotal)}</span>
          <button class="btn-icon" onclick="toggleSection('${section.id}')"
                  title="Collapse/Expand">▼</button>
          <button class="btn-icon" onclick="confirmDeleteSection('${section.id}')"
                  title="Delete Section">✕</button>
        </div>
        <div class="section-body" id="section-body-${section.id}">
          ${renderSectionLinesTable(section.id, sectionLines)}
          <div class="section-add-actions">
            <button class="btn btn-sm btn-secondary" onclick="addLineItemToSection('${section.id}')">
              + Add Line
            </button>
            <button class="btn btn-sm btn-secondary" onclick="openAssemblyPicker('${section.id}')">
              + Add Assembly
            </button>
          </div>
        </div>
      </div>
    `;
  }).join('');

  // Render unsectioned items
  const unsectionedContainer = document.getElementById('unsectionedContainer');
  if (unsectionedContainer) {

  setupSectionDragDrop();
}
```

```javascript
async function addSection() {
  if (!currentEstimate) return;

  const res = await fetch(`/api/estimates/${currentEstimate.id}/sections`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: 'New Section' })
  });

  if (!res.ok) {
    showToast('Failed to create section', 'error');
    return;
  }

  await reloadCurrentEstimate();
  showToast('Section added', 'success');
}

async function updateSectionName(sectionId, name) {
  if (!currentEstimate || !name.trim()) return;

  await fetch(`/api/estimates/${currentEstimate.id}/sections/${sectionId}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name: name.trim() })
  });

  // Update local state without full reload
  const section = currentEstimate.sections.find(s => s.id === sectionId);
  if (section) section.name = name.trim();
}

async function confirmDeleteSection(sectionId) {
  if (!confirm('Delete this section? Line items will be preserved but moved to "Other Items".')) {
    return;
  }

  await fetch(`/api/estimates/${currentEstimate.id}/sections/${sectionId}`, {
    method: 'DELETE'
  });

  await reloadCurrentEstimate();
  showToast('Section deleted', 'success');
}

function toggleSection(sectionId) {
  const body = document.getElementById(`section-body-${sectionId}`);
  if (body) {
    body.classList.toggle('collapsed');
    // Update toggle button icon
    const section = body.parentElement;
    const btn = section.querySelector('.section-header button');
    if (btn) btn.textContent = body.classList.contains('collapsed') ? '▶' : '▼';
  }
}
```

4. **Section drag-drop reordering:**
```javascript
function setupSectionDragDrop() {
  const container = document.getElementById('sectionsContainer');
  if (!container) return;

  const sections = container.querySelectorAll('.estimate-section');

  sections.forEach(section => {
    section.addEventListener('dragstart', handleSectionDragStart);
    section.addEventListener('dragend', handleSectionDragEnd);
    section.addEventListener('dragover', handleSectionDragOver);
    section.addEventListener('drop', handleSectionDrop);
  });
}

let draggedSection = null;

function handleSectionDragStart(e) {
  draggedSection = this;
  this.classList.add('dragging');
  e.dataTransfer.effectAllowed = 'move';
}

function handleSectionDragEnd(e) {
  this.classList.remove('dragging');
  document.querySelectorAll('.estimate-section').forEach(s => s.classList.remove('drag-over'));
  draggedSection = null;
}

function handleSectionDragOver(e) {
  e.preventDefault();
  this.classList.add('drag-over');
}

async function handleSectionDrop(e) {
  e.preventDefault();
  this.classList.remove('drag-over');

  if (!draggedSection || this === draggedSection) return;

  const container = document.getElementById('sectionsContainer');
  const sections = Array.from(container.querySelectorAll('.estimate-section'));
  const draggedIndex = sections.indexOf(draggedSection);
  const dropIndex = sections.indexOf(this);

  // Reorder in DOM
  if (draggedIndex < dropIndex) {
    this.parentNode.insertBefore(draggedSection, this.nextSibling);
  } else {
    this.parentNode.insertBefore(draggedSection, this);
  }

  // Persist new order
  const newOrder = Array.from(container.querySelectorAll('.estimate-section'))
    .map(s => s.dataset.sectionId);

  await fetch(`/api/estimates/${currentEstimate.id}/sections/reorder`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ section_ids: newOrder })
  });
}
```
