## Verification Script

```javascript
const SAVINGS = 'http://localhost:3001/api/savings';
const SPEND = 'http://localhost:3001/api/spend';

async function verify() {
  // Savings endpoints
  const summary = await fetch(`${SAVINGS}/summary`).then(r => r.json());
  console.log('Savings YTD:', '$' + summary.ytd?.savings_amount?.toFixed(2));

  const byJob = await fetch(`${SAVINGS}/by-job`).then(r => r.json());
  console.log('Jobs with savings:', byJob.length);

  const byPeriod = await fetch(`${SAVINGS}/by-period`).then(r => r.json());
  console.log('Monthly periods:', byPeriod.length);

  // Spend endpoints
  const vendorSpend = await fetch(`${SPEND}/by-vendor`).then(r => r.json());
  console.log('Vendors:', vendorSpend.vendor_count, 'Total:', '$' + vendorSpend.total_spend?.toFixed(2));

  const targets = await fetch(`${SPEND}/negotiation-targets`).then(r => r.json());
  console.log('Negotiation targets:', targets.targets?.length);
  console.log('Potential savings:', '$' + targets.summary?.total_potential_savings?.toFixed(2));

  const dashboard = await fetch(`${SPEND}/dashboard`).then(r => r.json());
  console.log('Dashboard YTD spend:', '$' + dashboard.summary?.ytd_spend?.toFixed(2));

  console.log('\n✓ All endpoints responding');
}

verify().catch(console.error);
```
