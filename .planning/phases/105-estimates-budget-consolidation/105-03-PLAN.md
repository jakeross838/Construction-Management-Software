   ```javascript
     // Check URL params first, then localStorage
     const urlParams = new URLSearchParams(window.location.search);
     const urlMode = urlParams.get('mode');

     if (urlMode === 'budget' || urlMode === 'estimates') {
       currentMode = urlMode;
       localStorage.setItem('estimatesBudgetMode', urlMode);
       // Clean up URL (remove mode param, keep others)
       urlParams.delete('mode');
       const newUrl = urlParams.toString()
         ? `${window.location.pathname}?${urlParams.toString()}`
         : window.location.pathname;
       window.history.replaceState({}, '', newUrl);
     } else {
       currentMode = localStorage.getItem('estimatesBudgetMode') || 'estimates';
     }

     // ... rest of initialization
   });
   ```

2. **Add URL state update when mode changes** in switchMode():
   ```javascript
   function switchMode(mode) {
     currentMode = mode;
     localStorage.setItem('estimatesBudgetMode', mode);

     // Update URL without triggering navigation
     const url = new URL(window.location);
     if (mode !== 'estimates') {
       url.searchParams.set('mode', mode);
     } else {
       url.searchParams.delete('mode');
     }
     window.history.pushState({}, '', url);

     updateModeUI();

     // Load data for the new mode
     if (mode === 'estimates') {
       loadEstimates();
       loadStats();
     } else if (mode === 'budget') {
       if (currentJobId) {
         loadJobBudgetForJob(currentJobId);
       } else {
         renderBudgetEmptyState();
       }
     }
   }
   ```

   ```javascript
   async function convertToBudget() {
     if (!currentEstimate) return;

     if (!confirm('Convert this estimate to budget lines? This will create or update budget entries for each cost code.')) return;

     try {
       const response = await fetch(`/api/estimates/${currentEstimate.id}/convert-to-budget`, {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify({ converted_by: 'User' })
       });

       if (!response.ok) {
         const err = await response.json();
         throw new Error(err.error || 'Failed to convert to budget');
       }

       const result = await response.json();

       // Store job ID for budget view
       const convertedJobId = currentEstimate.job_id;
       const jobName = currentEstimate.job?.name || 'this job';

       // Refresh estimate detail
       await openEstimateDetail(currentEstimate.id);
       await loadEstimates();
       await loadStats();

       // Show success with option to view budget
       showConversionSuccess(result.budget_lines, convertedJobId, jobName);

     } catch (err) {
       console.error('Error converting to budget:', err);
       showToast(err.message, 'error');
     }
   }
   ```

2. **Add conversion success modal/toast** with "View Budget" action:
   ```javascript
   function showConversionSuccess(lineCount, jobId, jobName) {
     // Create a more prominent notification with action button
     const notification = document.createElement('div');
     notification.className = 'conversion-success-notification';
     notification.innerHTML = `
       <div class="conversion-success-content">
         <div class="conversion-success-icon">✓</div>
         <div class="conversion-success-text">
           <strong>Budget Updated</strong>
           <p>${lineCount} budget line(s) created/updated for ${escapeHtml(jobName)}</p>
         </div>
         <div class="conversion-success-actions">
           <button class="btn btn-primary btn-sm" onclick="viewConvertedBudget('${jobId}'); this.closest('.conversion-success-notification').remove();">View Budget</button>
           <button class="btn btn-secondary btn-sm" onclick="this.closest('.conversion-success-notification').remove();">Dismiss</button>
         </div>
       </div>
     `;

     document.body.appendChild(notification);

     // Auto-dismiss after 10 seconds
     setTimeout(() => {
       if (notification.parentNode) {
         notification.remove();
       }
     }, 10000);
   }

   function viewConvertedBudget(jobId) {
     // Close the estimate detail modal
     closeDetailModal();

     // Set job in sidebar if available
     if (window.JobSidebar && jobId) {
       window.JobSidebar.setJob(jobId);
       currentJobId = jobId;
     }

     // Switch to budget mode
     switchMode('budget');
   }
   ```
