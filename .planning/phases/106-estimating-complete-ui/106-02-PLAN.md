# Phase 106-02: Estimating - Hierarchical API Routes

## Overview

Add API endpoints for managing hierarchical estimates: phases, groups, subgroups, and line items. Includes template management endpoints.

---

## Context

**New Tables (from 106-01):**
- `v2_estimate_phases` - Construction phases within estimate
- `v2_estimate_groups` - Groups within phases
- `v2_estimate_subgroups` - Subgroups within groups
- `v2_estimate_templates` - Reusable estimate structures

**Existing Route File:** `server/routes/estimates.js` (~2000 lines)

## Tasks

### Task 1: Add Hierarchical Estimate Fetch

**Goal:** Update GET /api/estimates/:id to include full hierarchy.

**Update in estimates.js:**
```javascript
// GET /api/estimates/:id - Get estimate with full hierarchy
router.get('/:id', async (req, res) => {
  try {
    const { id } = req.params;

    // Get estimate header
    const { data: estimate, error } = await supabase
      .from('v2_estimates')
      .select(`
        *,
        job:v2_jobs(id, name),
        created_by_user:created_by
      `)
      .eq('id', id)
      .single();

    if (error) throw error;
    if (!estimate) return res.status(404).json({ error: 'Estimate not found' });

    // Get phases with groups, subgroups, and line items
    const { data: phases } = await supabase
      .from('v2_estimate_phases')
      .select(`
        *,
        groups:v2_estimate_groups(
          *,
          subgroups:v2_estimate_subgroups(
            *,
            line_items:v2_estimate_line_items(
              *,
              cost_code:v2_cost_codes(id, code, name),
              catalog_item:v2_selection_catalog(id, name, unit_price, image_url)
            )
          )
        )
      `)
      .eq('estimate_id', id)
      .order('sort_order');

    // Sort nested arrays
    phases?.forEach(phase => {
      phase.groups?.sort((a, b) => a.sort_order - b.sort_order);
      phase.groups?.forEach(group => {
        group.subgroups?.sort((a, b) => a.sort_order - b.sort_order);
        group.subgroups?.forEach(subgroup => {
          subgroup.line_items?.sort((a, b) => a.sort_order - b.sort_order);
        });
      });
    });

    estimate.phases = phases || [];

    res.json(estimate);
  } catch (err) {
    console.error('Error fetching estimate:', err);
    res.status(500).json({ error: err.message });
  }
});
```

### Task 2: Add Phase CRUD Endpoints

**Goal:** Endpoints for managing estimate phases.

```javascript
// POST /api/estimates/:id/phases - Add phase
router.post('/:id/phases', async (req, res) => {
  try {
    const { id } = req.params;
    const { name, phase_code, description, sort_order } = req.body;

    // Get max sort_order if not provided
    let order = sort_order;
    if (order === undefined) {
      const { data: maxOrder } = await supabase
        .from('v2_estimate_phases')
        .select('sort_order')
        .eq('estimate_id', id)
        .order('sort_order', { ascending: false })
        .limit(1)
        .single();
      order = (maxOrder?.sort_order || 0) + 1;
    }

    const { data: phase, error } = await supabase
      .from('v2_estimate_phases')
      .insert({
        estimate_id: id,
        name,
        phase_code,
        description,
        sort_order: order
      })
      .select()
      .single();

    if (error) throw error;
    res.status(201).json(phase);
  } catch (err) {
    console.error('Error creating phase:', err);
    res.status(500).json({ error: err.message });
  }
});

// PATCH /api/estimates/phases/:phaseId - Update phase
```

### Task 5: Update Line Item Endpoints

**Goal:** Update line item endpoints to include subgroup_id.

```javascript
// POST /api/estimates/subgroups/:subgroupId/lines - Add line item to subgroup
router.post('/subgroups/:subgroupId/lines', async (req, res) => {
  try {
    const { subgroupId } = req.params;
    const {
      description,
      cost_code_id,
      catalog_item_id,
      quantity,
      unit,
      unit_cost,
      amount,
      notes,
      created_by
    } = req.body;

    // Get estimate_id from subgroup hierarchy
    const { data: subgroup } = await supabase
      .from('v2_estimate_subgroups')
      .select('group:v2_estimate_groups(phase:v2_estimate_phases(estimate_id))')
      .eq('id', subgroupId)
      .single();

    const estimate_id = subgroup?.group?.phase?.estimate_id;
    if (!estimate_id) {
      return res.status(404).json({ error: 'Subgroup not found' });
    }

    // Get max sort_order
    const { data: maxOrder } = await supabase
      .from('v2_estimate_line_items')
      .select('sort_order')
      .eq('subgroup_id', subgroupId)
      .order('sort_order', { ascending: false })
      .limit(1)
      .single();

    const sort_order = (maxOrder?.sort_order || 0) + 1;

    // Calculate amount if not provided
    const calculatedAmount = amount || (quantity * unit_cost);

    const { data: lineItem, error } = await supabase
      .from('v2_estimate_line_items')
      .insert({
        estimate_id,
        subgroup_id: subgroupId,
        description,
        cost_code_id,
        catalog_item_id,
        quantity: quantity || 1,
        unit,
        unit_cost: unit_cost || 0,
        amount: calculatedAmount,
        notes,
        sort_order,
        created_by
      })
      .select(`
        *,
        cost_code:v2_cost_codes(id, code, name),
        catalog_item:v2_selection_catalog(id, name, unit_price, image_url)
      `)
      .single();

    if (error) throw error;
    res.status(201).json(lineItem);
  } catch (err) {
    console.error('Error creating line item:', err);
    res.status(500).json({ error: err.message });
  }
});

// PATCH /api/estimates/lines/:lineId - Update line item
router.patch('/lines/:lineId', async (req, res) => {
  try {
    const { lineId } = req.params;
    const {
      subgroup_id,
      description,
      cost_code_id,
      catalog_item_id,
      quantity,
      unit,
      unit_cost,
      amount,
      notes,
      sort_order
    } = req.body;

    // Calculate amount if qty/unit_cost changed but amount not provided
    let calculatedAmount = amount;
    if (amount === undefined && (quantity !== undefined || unit_cost !== undefined)) {
      const { data: existing } = await supabase
        .from('v2_estimate_line_items')
        .select('quantity, unit_cost')
        .eq('id', lineId)
        .single();

      const newQty = quantity ?? existing?.quantity ?? 1;
      const newUnitCost = unit_cost ?? existing?.unit_cost ?? 0;
      calculatedAmount = newQty * newUnitCost;
    }

    const updateData = {
      ...(subgroup_id !== undefined && { subgroup_id }),
      ...(description !== undefined && { description }),
      ...(cost_code_id !== undefined && { cost_code_id }),
      ...(catalog_item_id !== undefined && { catalog_item_id }),
      ...(quantity !== undefined && { quantity }),
      ...(unit !== undefined && { unit }),
      ...(unit_cost !== undefined && { unit_cost }),
      ...(calculatedAmount !== undefined && { amount: calculatedAmount }),
      ...(notes !== undefined && { notes }),
      ...(sort_order !== undefined && { sort_order }),
      updated_at: new Date()
    };

    const { data: lineItem, error } = await supabase
      .from('v2_estimate_line_items')
      .update(updateData)
      .eq('id', lineId)
      .select(`
        *,
        cost_code:v2_cost_codes(id, code, name),
        catalog_item:v2_selection_catalog(id, name, unit_price, image_url)
      `)
      .single();

    if (error) throw error;
    res.json(lineItem);
  } catch (err) {
    console.error('Error updating line item:', err);
    res.status(500).json({ error: err.message });
  }
});
```

```javascript
// GET /api/estimates/templates - List templates
router.get('/templates', async (req, res) => {
  try {
    const { project_type, is_active } = req.query;

    let query = supabase
      .from('v2_estimate_templates')
      .select('*')
      .order('is_default', { ascending: false })
      .order('name');

    if (project_type) query = query.eq('project_type', project_type);
    if (is_active !== undefined) query = query.eq('is_active', is_active === 'true');

    const { data, error } = await query;
    if (error) throw error;

    res.json(data);
  } catch (err) {
    console.error('Error fetching templates:', err);
    res.status(500).json({ error: err.message });
  }
});

// GET /api/estimates/templates/:id - Get template details
router.get('/templates/:id', async (req, res) => {
  try {
    const { id } = req.params;

    const { data: template, error } = await supabase
      .from('v2_estimate_templates')
      .select('*')
      .eq('id', id)
      .single();

    if (error) throw error;
    if (!template) return res.status(404).json({ error: 'Template not found' });

    res.json(template);
  } catch (err) {
    console.error('Error fetching template:', err);
    res.status(500).json({ error: err.message });
  }

// POST /api/estimates/:id/apply-template - Apply template to estimate
router.post('/:id/apply-template', async (req, res) => {
  try {
    const { id } = req.params;
    const { template_id } = req.body;

    // Call database function
    const { error } = await supabase.rpc('apply_estimate_template', {
      p_estimate_id: id,
      p_template_id: template_id
    });

    if (error) throw error;

    // Return updated estimate with hierarchy
    const { data: estimate } = await supabase
      .from('v2_estimates')
      .select('*')
      .eq('id', id)
      .single();

    const { data: phases } = await supabase
      .from('v2_estimate_phases')
      .select(`
        *,
        groups:v2_estimate_groups(
          *,
          subgroups:v2_estimate_subgroups(*)
        )
      `)
      .eq('estimate_id', id)
      .order('sort_order');

    estimate.phases = phases || [];
    res.json(estimate);
  } catch (err) {
    console.error('Error applying template:', err);
    res.status(500).json({ error: err.message });
  }
});

// POST /api/estimates/:id/save-as-template - Save estimate as template
router.post('/:id/save-as-template', async (req, res) => {
  try {
    const { id } = req.params;
    const { name, description, created_by } = req.body;

    if (!name) {
      return res.status(400).json({ error: 'Template name is required' });
    }

    const { data: templateId, error } = await supabase.rpc('save_estimate_as_template', {
      p_estimate_id: id,
      p_name: name,
      p_description: description || null,
      p_created_by: created_by || 'User'
    });

    if (error) throw error;

    // Fetch the created template
    const { data: template } = await supabase
      .from('v2_estimate_templates')
      .select('*')
      .eq('id', templateId)
      .single();

    res.status(201).json(template);
  } catch (err) {
    console.error('Error saving template:', err);
    res.status(500).json({ error: err.message });
  }
});
```

### Task 7: Add Bulk Reorder Endpoint

**Goal:** Endpoint for drag-and-drop reordering.

```javascript
// POST /api/estimates/:id/reorder - Bulk reorder phases/groups/subgroups/items
router.post('/:id/reorder', async (req, res) => {
  try {
    const { id } = req.params;
    const { type, items } = req.body;
    // items = [{ id: 'uuid', sort_order: 1 }, ...]

    const tableMap = {
      phases: 'v2_estimate_phases',
      groups: 'v2_estimate_groups',
      subgroups: 'v2_estimate_subgroups',
      lines: 'v2_estimate_line_items'
    };

    const table = tableMap[type];
    if (!table) {
      return res.status(400).json({ error: 'Invalid type. Use: phases, groups, subgroups, or lines' });
    }

    // Update each item's sort_order
    for (const item of items) {
      await supabase
        .from(table)
        .update({ sort_order: item.sort_order, updated_at: new Date() })
        .eq('id', item.id);
    }

    res.json({ success: true, updated: items.length });
  } catch (err) {
    console.error('Error reordering:', err);
    res.status(500).json({ error: err.message });
  }
});
```
