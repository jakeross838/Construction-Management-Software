### Task 2: Create Learned Mappings Table

```sql
CREATE TABLE IF NOT EXISTS v2_cost_code_mappings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  description_pattern TEXT NOT NULL,
  cost_code_id UUID REFERENCES v2_cost_codes(id),
  vendor_trade TEXT,
  confidence DECIMAL(3,2) DEFAULT 0.80,
  usage_count INTEGER DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_ccm_pattern ON v2_cost_code_mappings(description_pattern);
CREATE INDEX idx_ccm_trade ON v2_cost_code_mappings(vendor_trade);
```

### Task 3: Refactor suggestCostCodes()

Update function to use database-driven approach:

```javascript
async function suggestCostCodes(lineItems, tradeType, jobId = null) {
  // 1. Load cost codes with keywords
  const { data: costCodes } = await supabase
    .from('v2_cost_codes')
    .select('id, code, name, category, keywords, trade_types');

  // 2. Load learned mappings for this trade
  const { data: mappings } = await supabase
    .from('v2_cost_code_mappings')
    .select('description_pattern, cost_code_id, confidence')
    .eq('vendor_trade', tradeType)
    .order('confidence', { ascending: false });

  // 3. For each line item, find best match
  return lineItems.map(item => {
    const desc = (item.description || '').toLowerCase();
    let matchedCode = null;
    let matchConfidence = 0;
    let matchReason = '';

    // Try learned mappings first (highest confidence)
    for (const mapping of (mappings || [])) {
      if (desc.includes(mapping.description_pattern.toLowerCase())) {
        matchedCode = costCodes.find(cc => cc.id === mapping.cost_code_id);
        matchConfidence = mapping.confidence;
        matchReason = 'learned_pattern';
        break;
      }
    }

    // Try cost code keywords
    if (!matchedCode) {
      for (const cc of costCodes) {
        if (cc.keywords?.some(kw => desc.includes(kw.toLowerCase()))) {
          matchedCode = cc;
          matchConfidence = 0.75;
          matchReason = 'keyword_match';
          break;
        }
      }
    }

    // Try trade type default
    if (!matchedCode && tradeType) {
      matchedCode = costCodes.find(cc => cc.trade_types?.includes(tradeType));
      if (matchedCode) {
        matchConfidence = 0.60;
        matchReason = 'trade_default';
      }
    }

    return {
      ...item,
      cost_code_id: matchedCode?.id || null,
      cost_code: matchedCode?.code || null,
      cost_code_name: matchedCode?.name || null,
      cost_code_confidence: matchConfidence,
      cost_code_match_reason: matchReason
    };
  });
}
```

### Task 4: Add Learning from Corrections

When user corrects a cost code assignment, save the mapping:

```javascript
async function learnCostCodeMapping(description, costCodeId, vendorTrade) {
  // Extract key phrase from description (first 50 chars, normalized)
  const pattern = description.toLowerCase().substring(0, 50).trim();

  // Upsert mapping
  const { error } = await supabase
    .from('v2_cost_code_mappings')
    .upsert({
      description_pattern: pattern,
      cost_code_id: costCodeId,
      vendor_trade: vendorTrade,
      confidence: 0.85,
      usage_count: 1,
      updated_at: new Date().toISOString()
    }, {
      onConflict: 'description_pattern,vendor_trade',
      ignoreDuplicates: false
    });

  if (!error) {
    // Increment usage count for existing
    await supabase.rpc('increment_mapping_usage', {
      pattern,
      trade: vendorTrade
    });
  }
}
```
