Create new `server/po-matcher.js` module with multi-signal PO matching.

**Signals and weights (from research):**
- Vendor similarity: 40% weight - Use existing fuzzyMatchScore or Jaro-Winkler
- PO number match: 25% weight - Exact or fuzzy match on extracted PO reference
- Amount proximity: 15% weight - Invoice amount within tolerance of PO remaining
- Date relevance: 10% weight - Invoice date within reasonable window of PO
- Line item match: 10% weight - Compare extracted line items to PO line items

**Core functions to implement:**

```javascript
/**
 * Calculate match score between invoice and PO candidate
 * @param {Object} invoice - { vendorName, amount, date, poRef, lineItems }
 * @param {Object} po - PO record with line items
 * @returns {Object} { score, breakdown, explanation }
 */
function calculateMatchScore(invoice, po)

/**
 * Find matching POs for an invoice with confidence scoring
 * @param {Object} invoiceData - Extracted invoice data
 * @param {string} jobId - Job ID to search within
 * @param {string} vendorId - Optional vendor ID (if already matched)
 * @returns {Object} { match, confidence, candidates, needsReview, explanation }
 */
async function findMatchingPO(invoiceData, jobId, vendorId)

/**
 * Calculate vendor similarity score
 * Uses existing fuzzyMatchScore from ai-processor.js
 */
function vendorSimilarity(invoiceVendor, poVendor)

/**
 * Check if PO number matches
 * Handles variations: PO-123, 123, PO123
 */
function poNumberMatch(invoicePoRef, poNumber)

/**
 * Calculate amount proximity score
 * @param tolerance - Default 0.10 (10%)
 */
function amountProximity(invoiceAmount, poRemaining, tolerance)

/**
 * Calculate date relevance score
 * @param windowDays - Default 90
 */
function dateRelevance(invoiceDate, poDate, windowDays)

/**
 * Compare line items between invoice and PO
 */
function lineItemMatch(invoiceItems, poItems)
```

**Match thresholds:**
- AUTO_MATCH: 0.95+ - High confidence, auto-assign
- REVIEW_QUEUE: 0.80-0.95 - Good match but needs review
- INVESTIGATE: 0.60-0.80 - Possible match, needs human verification
- NO_MATCH: below 0.60 - No suitable match found

**Export structure:**
```javascript
module.exports = {
  findMatchingPO,
  calculateMatchScore,
  MATCH_THRESHOLDS: { AUTO_MATCH: 0.95, REVIEW_QUEUE: 0.80, INVESTIGATE: 0.60 }
};
```

async function findOrCreatePO(jobId, vendorId, invoiceData, jobName) {
  if (!jobId) return null;

  // Use new multi-signal matching
  const matchResult = await poMatcher.findMatchingPO(invoiceData, jobId, vendorId);

  if (matchResult.match) {
    return {
      po: matchResult.match,
      isNew: false,
      matchConfidence: matchResult.confidence,
      matchBreakdown: matchResult.breakdown,
      needsReview: matchResult.needsReview,
      explanation: matchResult.explanation,
      candidates: matchResult.candidates
    };
  }

  // No auto-create - return match info for review
  return {
    po: null,
    isNew: false,
    matchConfidence: 0,
    candidates: matchResult.candidates,
    needsReview: matchResult.candidates.length > 0,
    explanation: matchResult.explanation
  };
}
