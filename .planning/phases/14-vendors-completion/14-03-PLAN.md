# Plan 14-03: Duplicate Detection Enhancement

<objective>
Add real-time duplicate detection when creating vendors, warning users before they create a potential duplicate. This fulfills VND-03.
</objective>

## Context

**Existing state:**
- `/api/vendors/duplicates` - returns list of existing duplicate pairs
- `/api/vendors/merge` - merges two vendors
- `findPotentialDuplicateVendors()` in ai-learning.js does similarity matching
- Frontend shows duplicates warning in stats bar after vendors load

**Gap:**
- No warning when creating a new vendor
- User can create duplicate, then has to merge later
- No "check for duplicates" API endpoint for a single name

### Task 1: Add duplicate check endpoint
**File:** `server/routes/vendors.js`

Add endpoint to check if a vendor name has potential duplicates:

```javascript
// Check for similar vendors (used before create)
router.get('/check-duplicate', asyncHandler(async (req, res) => {
  const { name, threshold = 75 } = req.query;

  if (!name || name.trim().length < 2) {
    return res.json({ matches: [] });
  }

  const searchName = name.trim().toLowerCase();

  // Get all active vendors
  const { data: vendors, error } = await supabase
    .from('v2_vendors')
    .select('id, name, email, phone, trade')
    .is('deleted_at', null);

  if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });

  // Find similar vendors using string similarity
  const matches = [];
  for (const vendor of vendors || []) {
    const similarity = calculateSimilarity(searchName, (vendor.name || '').toLowerCase());
    if (similarity >= threshold) {
      matches.push({
        ...vendor,
        similarity: Math.round(similarity)
      });
    }
  }

  // Sort by similarity descending
  matches.sort((a, b) => b.similarity - a.similarity);

  res.json({
    query: name,
    threshold,
    matches: matches.slice(0, 5) // Top 5 matches
  });
}));

// String similarity using Levenshtein-based algorithm
function calculateSimilarity(str1, str2) {
  if (str1 === str2) return 100;
  if (!str1 || !str2) return 0;

  // Check for common abbreviation patterns
  const normalized1 = normalizeVendorName(str1);
  const normalized2 = normalizeVendorName(str2);
  if (normalized1 === normalized2) return 95;

  // Check if one contains the other
  if (normalized1.includes(normalized2) || normalized2.includes(normalized1)) {
    const ratio = Math.min(normalized1.length, normalized2.length) / Math.max(normalized1.length, normalized2.length);
    return Math.round(70 + ratio * 25);
  }

  // Levenshtein distance
  const maxLen = Math.max(str1.length, str2.length);
  if (maxLen === 0) return 100;

  const distance = levenshteinDistance(str1, str2);
  return Math.round((1 - distance / maxLen) * 100);
}

function normalizeVendorName(name) {
  return name
    .toLowerCase()
    .replace(/\b(inc|llc|corp|company|co|ltd|services|service)\b\.?/gi, '')
    .replace(/[^a-z0-9]/g, '')
    .trim();
}

function levenshteinDistance(str1, str2) {
  const m = str1.length;
  const n = str2.length;
  const dp = Array(m + 1).fill(null).map(() => Array(n + 1).fill(0));

  for (let i = 0; i <= m; i++) dp[i][0] = i;
  for (let j = 0; j <= n; j++) dp[0][j] = j;

  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (str1[i - 1] === str2[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1];
      } else {
        dp[i][j] = 1 + Math.min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]);
      }
    }
  }
  return dp[m][n];
}
```

### Task 2: Update POST /api/vendors to warn on duplicates
**File:** `server/routes/vendors.js`

Update the create endpoint to return duplicate warnings:

```javascript
// Create vendor (with duplicate warning)
router.post('/', validateRequest({ body: { name: { required: true } } }), asyncHandler(async (req, res) => {
  const { name, skip_duplicate_check } = req.body;

  // Check for duplicates unless explicitly skipped
  if (!skip_duplicate_check) {
    const searchName = name.trim().toLowerCase();
    const { data: vendors } = await supabase
      .from('v2_vendors')
      .select('id, name, email, phone, trade')
      .is('deleted_at', null);

    const duplicates = [];
    for (const vendor of vendors || []) {
      const similarity = calculateSimilarity(searchName, (vendor.name || '').toLowerCase());
      if (similarity >= 75) {
        duplicates.push({ ...vendor, similarity: Math.round(similarity) });
      }
    }

    if (duplicates.length > 0) {
      return res.status(409).json({
        error: 'DUPLICATE_WARNING',
        message: 'Similar vendor(s) already exist',
        duplicates: duplicates.sort((a, b) => b.similarity - a.similarity).slice(0, 3),
        hint: 'Set skip_duplicate_check: true to create anyway'
      });
    }
  }

  // Create the vendor
  const { skip_duplicate_check: _, ...vendorData } = req.body;
  const { data, error } = await supabase
    .from('v2_vendors')
    .insert(vendorData)
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });
  res.json(data);
}));
```

### Task 3: Add duplicate warning UI to vendor modal
**File:** `public/vendors.html`

Add real-time duplicate checking when typing vendor name:

```javascript
// Add duplicate warning container to modal
<div id="duplicateWarningBanner" class="duplicate-warning-banner" style="display: none;">
  <div class="warning-icon">⚠️</div>
  <div class="warning-content">
    <span class="warning-title">Similar vendor found</span>
    <div id="duplicateMatches" class="duplicate-matches"></div>
  </div>
  <button class="btn btn-sm btn-link" onclick="dismissDuplicateWarning()">Dismiss</button>
</div>

// Add debounced duplicate check on name input
let duplicateCheckTimer;
document.getElementById('vendorName').addEventListener('input', (e) => {
  clearTimeout(duplicateCheckTimer);
  const name = e.target.value.trim();

  // Only check if name is long enough and we're creating (not editing)
  if (name.length < 3 || document.getElementById('vendorId').value) {
    hideDuplicateWarning();
    return;
  }

  duplicateCheckTimer = setTimeout(() => checkForDuplicates(name), 500);
});

async function checkForDuplicates(name) {
  try {
    const res = await fetch(`/api/vendors/check-duplicate?name=${encodeURIComponent(name)}`);
    const result = await res.json();

    if (result.matches && result.matches.length > 0) {
      showDuplicateWarning(result.matches);
    } else {
      hideDuplicateWarning();
    }
  } catch (err) {
    console.error('Duplicate check failed:', err);
  }
}

function showDuplicateWarning(matches) {
  const banner = document.getElementById('duplicateWarningBanner');
  const matchesDiv = document.getElementById('duplicateMatches');

  matchesDiv.innerHTML = matches.map(m => `
    <div class="duplicate-match">
      <span class="match-name">${m.name}</span>
      <span class="match-similarity">${m.similarity}% match</span>
      <button class="btn btn-xs btn-link" onclick="useExistingVendor('${m.id}')">Use this vendor</button>
    </div>
  `).join('');

  banner.style.display = 'flex';
}

function hideDuplicateWarning() {
  document.getElementById('duplicateWarningBanner').style.display = 'none';
}

function dismissDuplicateWarning() {
  hideDuplicateWarning();
  // Mark that user dismissed the warning
  state.duplicateWarningDismissed = true;
}

function useExistingVendor(vendorId) {
  closeVendorModal();
  openVendor(vendorId);
}

// Update saveVendor to handle duplicate response
async function saveVendor() {
  const id = document.getElementById('vendorId').value;
  const data = { /* ... existing data collection ... */ };

  // If user dismissed warning, skip check
  if (state.duplicateWarningDismissed && !id) {
    data.skip_duplicate_check = true;
  }

  try {
    const url = id ? `/api/vendors/${id}` : '/api/vendors';
    const method = id ? 'PATCH' : 'POST';

    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });

    const result = await res.json();

    // Handle duplicate warning from server
    if (res.status === 409 && result.error === 'DUPLICATE_WARNING') {
      showDuplicateWarning(result.duplicates);
      showToast('Similar vendor already exists - check the warning', 'warning');
      return;
    }

    if (!res.ok) throw new Error(result.error || 'Failed to save vendor');

    showToast(id ? 'Vendor updated' : 'Vendor created', 'success');
    state.duplicateWarningDismissed = false;
    closeVendorModal();
    await loadVendors();
    await checkDuplicates();
  } catch (err) {
    showToast(err.message, 'error');
  }
}
```
