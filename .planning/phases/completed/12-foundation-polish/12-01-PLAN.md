       - Add `app.use(errorMiddleware)` at END of all routes (after all app.use('/api/...'))

    2. In server/errors.js, add a validateRequest helper:
       ```javascript
       /**
        * Validate request body/params against a schema
        * @param {Object} schema - { body?: {...}, params?: {...} }
        * Returns middleware that validates and throws AppError on failure
        */
       function validateRequest(schema) {
         return (req, res, next) => {
           const errors = [];

           if (schema.body) {
             for (const [field, rules] of Object.entries(schema.body)) {
               const value = req.body?.[field];
               if (rules.required && (value === undefined || value === null || value === '')) {
                 errors.push({ field, message: `${field} is required` });
               }
               if (rules.type === 'uuid' && value && !/^[0-9a-f-]{36}$/i.test(value)) {
                 errors.push({ field, message: `${field} must be a valid UUID` });
               }
               if (rules.type === 'number' && value !== undefined && isNaN(Number(value))) {
                 errors.push({ field, message: `${field} must be a number` });
               }
             }
           }

           if (schema.params) {
             for (const [field, rules] of Object.entries(schema.params)) {
               const value = req.params?.[field];
               if (rules.required && !value) {
                 errors.push({ field, message: `${field} param is required` });
               }
               if (rules.type === 'uuid' && value && !/^[0-9a-f-]{36}$/i.test(value)) {
                 errors.push({ field, message: `${field} must be a valid UUID` });
               }
             }
           }

           if (errors.length > 0) {
             throw new AppError('VALIDATION_FAILED', 'Request validation failed', { fields: errors });
           }
           next();
         };
       }
       ```
       Export validateRequest.

    For each route file that doesn't use asyncHandler (12 files):

    1. Add imports at top:
       ```javascript
       const { asyncHandler, AppError, notFoundError } = require('../errors');
       ```

    2. Wrap all route handlers with asyncHandler:
       - Change: `router.get('/', async (req, res) => {`
       - To: `router.get('/', asyncHandler(async (req, res) => {`
       - Add closing `)` at end of handler

    3. Remove try/catch blocks - asyncHandler handles errors

    4. Replace `res.status(500).json({ error: err.message })` patterns with:
       - Let error propagate to errorMiddleware (just throw)

    5. Replace `res.status(404).json({ error: 'X not found' })` patterns with:
       - `throw notFoundError('entity', id)` or `throw new AppError('CODE', 'message', { details })`

    Example transformation (vendors.js):
    BEFORE:
    ```javascript
    router.get('/', async (req, res) => {
      try {
        const { data, error } = await supabase.from('v2_vendors').select('*');
        if (error) throw error;
        res.json(data);
      } catch (err) {
        res.status(500).json({ error: err.message });
      }
    });
    ```

    AFTER:
    ```javascript
    router.get('/', asyncHandler(async (req, res) => {
      const { data, error } = await supabase.from('v2_vendors').select('*');
      if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });
      res.json(data);
    }));
    ```
