# Phase 106-04: Estimating - Catalog Integration

## Overview

Integrate the selection catalog into estimate line items. When adding items, show catalog suggestions based on subgroup context. Pull pricing, labor hours, and other data from catalog items.

---

## Context

**Dependencies:**
- 106-01: Schema with hierarchy
- 106-02: API endpoints
- 106-03: UI rendering

**Catalog System (from v3.0):**
- `v2_selection_catalog` - Product catalog with pricing
- `v2_selection_categories` - Category hierarchy
- `v2_catalog_trades` - Trade associations with labor hours
- Items have: `unit_price`, `labor_hours`, `unit`, `coverage_rate`

**Integration Points:**
- When adding line item, search catalog by subgroup name/category
- Show matching catalog items as suggestions
- Auto-fill pricing and labor data from selected catalog item
- Link line item to catalog item for price updates

### Task 1: Add Catalog Search Endpoint

**Goal:** API to search catalog items relevant to a subgroup.

**Add to estimates.js:**
```javascript
// GET /api/estimates/catalog-suggestions?subgroup_id=xxx&query=xxx
router.get('/catalog-suggestions', async (req, res) => {
  try {
    const { subgroup_id, query } = req.query;

    // Get subgroup context (name for matching)
    let contextTerms = [];
    if (subgroup_id) {
      const { data: subgroup } = await supabase
        .from('v2_estimate_subgroups')
        .select(`
          name,
          group:v2_estimate_groups(
            name,
            phase:v2_estimate_phases(name, phase_code)
          )
        `)
        .eq('id', subgroup_id)
        .single();

      if (subgroup) {
        contextTerms = [
          subgroup.name,
          subgroup.group?.name,
          subgroup.group?.phase?.name
        ].filter(Boolean);
      }
    }

    // Build search query
    let catalogQuery = supabase
      .from('v2_selection_catalog')
      .select(`
        id,
        name,
        description,
        unit_price,
        unit,
        labor_hours,
        material,
        brand,
        image_url,
        category:v2_selection_categories(id, name)
      `)
      .limit(20);

    // Search by query or context
    const searchTerm = query || contextTerms.join(' ');
    if (searchTerm) {
      catalogQuery = catalogQuery.or(`name.ilike.%${searchTerm}%,description.ilike.%${searchTerm}%,tags.cs.{${searchTerm.toLowerCase()}}`);
    }

    const { data: items, error } = await catalogQuery;
    if (error) throw error;

    // Sort by relevance (context matches first)
    const scored = items.map(item => {
      let score = 0;
      const itemText = `${item.name} ${item.description || ''} ${item.category?.name || ''}`.toLowerCase();

      contextTerms.forEach(term => {
        if (itemText.includes(term.toLowerCase())) score += 10;
      });

      if (query && itemText.includes(query.toLowerCase())) score += 5;

      return { ...item, relevance: score };
    });

    scored.sort((a, b) => b.relevance - a.relevance);

    res.json(scored.slice(0, 10));
  } catch (err) {
    console.error('Error fetching catalog suggestions:', err);
    res.status(500).json({ error: err.message });
  }
});
```

### Task 3: Implement Catalog Search/Selection JavaScript

**Goal:** Search catalog and auto-fill line item fields.

```javascript
let catalogSearchDebounce;
let selectedCatalogItem = null;

async function searchCatalog(query) {
  clearTimeout(catalogSearchDebounce);
  catalogSearchDebounce = setTimeout(async () => {
    const subgroupId = currentSubgroupId; // Set when opening modal
    await loadCatalogSuggestions(subgroupId, query);
  }, 300);
}

async function loadCatalogSuggestions(subgroupId, query = '') {
  const container = document.getElementById('catalogSuggestions');

  try {
    const params = new URLSearchParams();
    if (subgroupId) params.set('subgroup_id', subgroupId);
    if (query) params.set('query', query);

    const response = await fetch(`/api/estimates/catalog-suggestions?${params}`);
    const items = await response.json();

    if (items.length === 0) {
      container.innerHTML = `
        <div class="catalog-suggestion" style="justify-content: center; color: var(--text-secondary);">
          No matching items found
        </div>
      `;
      return;
    }

    container.innerHTML = items.map(item => `
      <div class="catalog-suggestion" onclick="selectCatalogItem('${item.id}', ${JSON.stringify(item).replace(/"/g, '&quot;')})">
        <img class="catalog-suggestion-image"
             src="${item.image_url || '/images/placeholder-product.png'}"
             alt="${escapeHtml(item.name)}">
        <div class="catalog-suggestion-info">
          <div class="catalog-suggestion-name">${escapeHtml(item.name)}</div>
          <div class="catalog-suggestion-meta">
            ${item.category?.name || ''} ${item.brand ? `• ${item.brand}` : ''}
          </div>
        </div>
        <div class="catalog-suggestion-price">${formatCurrency(item.unit_price)}/${item.unit || 'ea'}</div>
      </div>
    `).join('');
  } catch (err) {
    console.error('Error loading catalog suggestions:', err);
    container.innerHTML = `
      <div class="catalog-suggestion" style="color: var(--accent-red);">
        Error loading suggestions
      </div>
    `;
  }
}

function selectCatalogItem(itemId, itemData) {
  selectedCatalogItem = itemData;

  // Store the ID
  document.getElementById('lineCatalogItemId').value = itemId;

  // Auto-fill form fields
  document.getElementById('lineDescription').value = itemData.name;
  document.getElementById('lineUnitCost').value = itemData.unit_price || 0;
  document.getElementById('lineUnit').value = itemData.unit || 'ea';

  // If quantity is still 1, keep it; otherwise respect user's entry
  const qtyField = document.getElementById('lineQuantity');
  if (!qtyField.value || qtyField.value === '1') {
    qtyField.value = 1;
  }

  // Calculate amount
  calculateLineAmount();

  // Hide suggestions, show selected
  document.getElementById('catalogSuggestions').style.display = 'none';
  const selectedDiv = document.getElementById('selectedCatalogItem');
  selectedDiv.style.display = 'flex';
  selectedDiv.innerHTML = `
    <img class="catalog-suggestion-image"
         src="${itemData.image_url || '/images/placeholder-product.png'}"
         alt="${escapeHtml(itemData.name)}">
    <div class="catalog-suggestion-info">
      <div class="catalog-suggestion-name">${escapeHtml(itemData.name)}</div>
      <div class="catalog-suggestion-meta">
        ${itemData.category?.name || ''} • ${formatCurrency(itemData.unit_price)}/${itemData.unit || 'ea'}
      </div>
    </div>
    <span class="catalog-badge">From Catalog</span>
  `;

  showToast(`Selected: ${itemData.name}`, 'success');
}

function clearCatalogSelection() {
  selectedCatalogItem = null;
  document.getElementById('lineCatalogItemId').value = '';
  document.getElementById('selectedCatalogItem').style.display = 'none';
  document.getElementById('catalogSuggestions').style.display = 'block';
  document.getElementById('catalogSearch').value = '';

  // Reload context-based suggestions
  loadCatalogSuggestions(currentSubgroupId);
}
```

```javascript
async function saveLineItem() {
  if (!currentEstimate || !currentSubgroupId) {
    showToast('No subgroup selected', 'error');
    return;
  }

  const lineId = document.getElementById('lineItemId').value;
  const catalogItemId = document.getElementById('lineCatalogItemId').value || null;

  const data = {
    description: document.getElementById('lineDescription').value,
    cost_code_id: document.getElementById('lineCostCode').value || null,
    catalog_item_id: catalogItemId,
    quantity: parseFloat(document.getElementById('lineQuantity').value) || 1,
    unit: document.getElementById('lineUnit').value || null,
    unit_cost: parseFloat(document.getElementById('lineUnitCost').value) || 0,
    amount: parseFloat(document.getElementById('lineAmount').value) || 0,
    notes: document.getElementById('lineNotes').value || null,
    created_by: window.currentUser || 'User'
  };

  if (!data.description) {
    showToast('Description is required', 'error');
    return;
  }

  try {
    let url, method;

    if (lineId) {
      // Editing existing line
      url = `/api/estimates/lines/${lineId}`;
      method = 'PATCH';
    } else {
      // Adding new line to subgroup
      url = `/api/estimates/subgroups/${currentSubgroupId}/lines`;
      method = 'POST';
    }

    const response = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });

    if (!response.ok) {
      const err = await response.json();
      throw new Error(err.error || 'Failed to save line item');
    }

    showToast(lineId ? 'Line item updated' : 'Line item added', 'success');
    closeLineModal();
    await refreshEstimate();
  } catch (err) {
    console.error('Error saving line item:', err);
    showToast(err.message, 'error');
  }
}
```

### Task 7: Add "Browse Catalog" Button

**Goal:** Full catalog browse modal for complex selection.

```javascript
async function openCatalogBrowser() {
  const modal = document.getElementById('catalogBrowserModal');

  // Load categories
  const categories = await loadCatalogCategories();
  renderCatalogCategories(categories);

  // Load all items initially
  await loadCatalogBrowserItems();

  modal.style.display = 'flex';
  modal.classList.add('show');
}

async function loadCatalogCategories() {
  const response = await fetch('/api/selections/categories');
  return response.json();
}

function renderCatalogCategories(categories) {
  const container = document.getElementById('catalogCategoryList');
  container.innerHTML = `
    <div class="catalog-category active" onclick="filterCatalogByCategory(null)">
      All Items
    </div>
    ${categories.map(cat => `
      <div class="catalog-category" onclick="filterCatalogByCategory('${cat.id}')">
        ${escapeHtml(cat.name)}
        <span class="category-count">${cat.item_count || 0}</span>
      </div>
    `).join('')}
  `;
}

async function loadCatalogBrowserItems(categoryId = null, search = '') {
  const container = document.getElementById('catalogBrowserItems');
  container.innerHTML = '<div class="loading-state"><div class="spinner"></div></div>';

  try {
    let url = '/api/selections/catalog?limit=50';
    if (categoryId) url += `&category_id=${categoryId}`;
    if (search) url += `&search=${encodeURIComponent(search)}`;

    const response = await fetch(url);
    const items = await response.json();

    if (items.length === 0) {
      container.innerHTML = `
        <div class="empty-state">
          <p>No items found</p>
        </div>
      `;
      return;
    }

    container.innerHTML = `
      <div class="catalog-grid">
        ${items.map(item => `
          <div class="catalog-card" onclick="selectFromBrowser('${item.id}', ${JSON.stringify(item).replace(/"/g, '&quot;')})">
            <img src="${item.image_url || '/images/placeholder-product.png'}" alt="${escapeHtml(item.name)}">
            <div class="catalog-card-info">
              <h4>${escapeHtml(item.name)}</h4>
              <p>${escapeHtml(item.brand || item.category?.name || '')}</p>
              <span class="catalog-card-price">${formatCurrency(item.unit_price)}/${item.unit || 'ea'}</span>
            </div>
          </div>
        `).join('')}
      </div>
    `;
  } catch (err) {
    console.error('Error loading catalog items:', err);
    container.innerHTML = '<div class="empty-state"><p>Error loading catalog</p></div>';
  }
}

function selectFromBrowser(itemId, itemData) {
  // Close browser modal
  closeCatalogBrowser();

  // Select the item (reuse existing function)
  selectCatalogItem(itemId, itemData);
}
```
