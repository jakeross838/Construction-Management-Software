Create `server/routes/assembly-templates.js` with full CRUD for assembly templates:

1. **GET /api/assembly-templates** - List templates with filters:
   - Query params: `category`, `is_active` (boolean), `search` (name search)
   - Returns: Array of templates with `item_count` computed field
   - Include items in response when fetching single template

2. **GET /api/assembly-templates/:id** - Get single template with all items:
   - JOIN v2_assembly_template_items ordered by sort_order
   - Compute `total_amount` (sum of qty * unit_cost for items)

3. **POST /api/assembly-templates** - Create template:
   - Required: `name`
   - Optional: `description`, `category`, `is_active` (default true), `created_by`
   - Return created template

4. **PATCH /api/assembly-templates/:id** - Update template:
   - Can update: `name`, `description`, `category`, `is_active`
   - Set `updated_at = NOW()`

5. **DELETE /api/assembly-templates/:id** - Soft delete (set is_active = false):
   - Don't actually delete - templates may be referenced by existing estimates
   - Return success message

6. **Template Items sub-routes:**
   - POST `/api/assembly-templates/:id/items` - Add item to template
   - PATCH `/api/assembly-templates/:id/items/:itemId` - Update item
   - DELETE `/api/assembly-templates/:id/items/:itemId` - Remove item (hard delete OK)
   - POST `/api/assembly-templates/:id/items/reorder` - Reorder items (accepts `item_ids` array)

7. **GET /api/assembly-templates/categories** - List distinct categories for filter dropdown

Pattern to follow from existing routes:
```javascript
const express = require('express');
const router = express.Router();
const { supabase } = require('../../config');
const { AppError, asyncHandler } = require('../errors');

// List templates
router.get('/', asyncHandler(async (req, res) => {
  const { category, is_active, search } = req.query;

  let query = supabase
    .from('v2_assembly_templates')
    .select('*, items:v2_assembly_template_items(count)');

  if (category) query = query.eq('category', category);
  if (is_active !== undefined) query = query.eq('is_active', is_active === 'true');
  if (search) query = query.ilike('name', `%${search}%`);

  const { data, error } = await query.order('name');
  if (error) throw new AppError('DATABASE_ERROR', error.message);

  res.json(data);
}));
```

```javascript
async function createTemplate() {
  const res = await fetch('/api/assembly-templates', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      name: 'New Assembly Template',
      category: 'Uncategorized',
      is_active: true,
      created_by: window.currentUser || 'User'
    })
  });
  const template = await res.json();
  await loadTemplates();
  openTemplateEditor(template.id);
  showToast('Template created', 'success');
}

async function saveTemplate() {
  const template = {
    name: document.getElementById('templateName').value,
    description: document.getElementById('templateDescription').value,
    category: document.getElementById('templateCategory').value,
    is_active: document.getElementById('templateActive').checked
  };

  await fetch(`/api/assembly-templates/${currentTemplateId}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(template)
  });

  await loadTemplates();
  closeTemplateEditor();
  showToast('Template saved', 'success');
}

async function deleteTemplate() {
  if (!confirm('Deactivate this template? It will no longer appear in the assembly picker.')) return;

  await fetch(`/api/assembly-templates/${currentTemplateId}`, { method: 'DELETE' });
  await loadTemplates();
  closeTemplateEditor();
  showToast('Template deactivated', 'success');
}
```
