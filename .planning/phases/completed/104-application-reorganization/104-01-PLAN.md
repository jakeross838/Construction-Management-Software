Add a detectPageContext() function that determines whether the current page is job-context or company-context.

```javascript
// Company-context pages (no job sidebar)
const COMPANY_CONTEXT_PAGES = [
  'dashboard.html',
  'business-dashboard.html',
  'catalog.html',
  'vendors.html',
  'cost-codes.html',
  'employees.html',
  'crew-schedule.html',
  'timesheets.html',
  'companies.html',
  'contacts.html',
  'price-intelligence.html',
  'expenses.html',
  'financial-periods.html',
  'overhead.html',
  'profitability.html',
  'wip.html',
  'pnl.html',
  'cash-flow.html',
  'business-planning.html'
];

function detectPageContext() {
  // Check data attribute first (highest priority)
  const bodyContext = document.body.dataset.pageContext;
  if (bodyContext === 'job' || bodyContext === 'company') {
    return bodyContext;
  }

  // Fall back to URL matching
  const path = window.location.pathname;
  const filename = path.split('/').pop() || 'index.html';

  return COMPANY_CONTEXT_PAGES.includes(filename) ? 'company' : 'job';
}
```
