### Task 1: Add vendor soft delete endpoint to vendors.js
**File:** `server/routes/vendors.js`

Add DELETE route:
```javascript
// Soft delete vendor
router.delete('/:id', asyncHandler(async (req, res) => {
  const { id } = req.params;

  // Check vendor exists and not already deleted
  const { data: vendor, error: findError } = await supabase
    .from('v2_vendors')
    .select('id, name, deleted_at')
    .eq('id', id)
    .single();

  if (findError || !vendor) {
    throw notFoundError('vendor', id);
  }

  if (vendor.deleted_at) {
    throw new AppError('ALREADY_DELETED', 'Vendor is already deleted', { vendor_id: id });
  }

  // Soft delete
  const { data, error } = await supabase
    .from('v2_vendors')
    .update({ deleted_at: new Date().toISOString() })
    .eq('id', id)
    .select()
    .single();

  if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });

  res.json({ success: true, deleted: data });
}));
```

### Task 2: Update GET /api/vendors to filter deleted vendors
**File:** `server/routes/vendors.js`

Modify GET route to exclude deleted:
```javascript
router.get('/', asyncHandler(async (req, res) => {
  const { search, trade, include_deleted } = req.query;

  let query = supabase
    .from('v2_vendors')
    .select('*');

  // Filter deleted unless explicitly requested
  if (include_deleted !== 'true') {
    query = query.is('deleted_at', null);
  }

  // Server-side search
  if (search) {
    query = query.or(`name.ilike.%${search}%,email.ilike.%${search}%,phone.ilike.%${search}%`);
  }

  // Trade filter
  if (trade) {
    query = query.eq('trade', trade);
  }

  const { data, error } = await query.order('name');

  if (error) throw new AppError('DATABASE_ERROR', error.message, { code: error.code });
  res.json(data);
}));
```

// In loadVendors function
async function loadVendors() {
  try {
    const params = new URLSearchParams();
    if (state.searchQuery) params.set('search', state.searchQuery);

// Add delete function
async function deleteCurrentVendor() {
  if (!state.currentVendor) return;

  const confirmed = confirm(`Are you sure you want to delete "${state.currentVendor.name}"?\n\nThis will archive the vendor but preserve all associated invoices and POs.`);
  if (!confirmed) return;

  try {
    const res = await fetch(`/api/vendors/${state.currentVendor.id}`, {
      method: 'DELETE'
    });

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.error || 'Failed to delete vendor');
    }

    showToast('Vendor deleted', 'success');
    closeVendorDetailModal();
    await loadVendors();
  } catch (err) {
    showToast(err.message, 'error');
  }
}
