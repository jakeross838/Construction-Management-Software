```javascript
/**
 * Similarity ratio (0-1) based on Levenshtein
 */
function similarityRatio(str1, str2) {
  if (!str1 || !str2) return 0;
  const maxLen = Math.max(str1.length, str2.length);
  if (maxLen === 0) return 1;
  const distance = levenshteinDistance(str1, str2);
  return 1 - (distance / maxLen);
}

/**
 * Normalize address for comparison
 */
function normalizeAddress(address) {
  if (!address) return '';
  return address.toLowerCase()
    .replace(/\b(street|st)\b/g, 'st')
    .replace(/\b(avenue|ave)\b/g, 'ave')
    .replace(/\b(road|rd)\b/g, 'rd')
    .replace(/\b(drive|dr)\b/g, 'dr')
    .replace(/\b(lane|ln)\b/g, 'ln')
    .replace(/\b(court|ct)\b/g, 'ct')
    .replace(/[.,#]/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

/**
 * Extract street number from address
 */
function extractStreetNumber(address) {
  const match = address?.match(/^(\d+)/);
  return match ? match[1] : null;
}

module.exports = { levenshteinDistance, similarityRatio, normalizeAddress, extractStreetNumber };
```

### Task 2: Enhanced findMatchingJob()

Refactor to use fuzzy matching:

```javascript
async function findMatchingJob(jobData) {
  if (!jobData) return null;

  const { data: jobs, error } = await supabase
    .from('v2_jobs')
    .select('id, name, address, client_name, status')
    .in('status', ['active', 'on_hold']);  // Include on_hold jobs too

  if (error || !jobs?.length) return null;

  const candidates = [];
  const searchAddress = normalizeAddress(jobData.address);
  const searchClient = (jobData.clientName || '').toLowerCase();
  const searchRef = (jobData.reference || '').toLowerCase();

  for (const job of jobs) {
    const jobName = (job.name || '').toLowerCase();
    const jobAddress = normalizeAddress(job.address);
    const jobClient = (job.client_name || '').toLowerCase();
    const clientFromName = jobName.split(/[-–\s]+/)[0];

    let score = 0;
    let matchStrategy = '';

    // Strategy 1: Exact client name match (highest)
    if (searchClient && (jobClient === searchClient || clientFromName === searchClient)) {
      score = 0.95;
      matchStrategy = 'exact_client';
    }

    // Strategy 2: Address number + street match
    if (!score && searchAddress && jobAddress) {
      const searchNum = extractStreetNumber(searchAddress);
      const jobNum = extractStreetNumber(jobAddress);

      if (searchNum && jobNum && searchNum === jobNum) {
        // Same street number - check street name similarity
        const streetSim = similarityRatio(searchAddress, jobAddress);
        if (streetSim > 0.7) {
          score = 0.90;
          matchStrategy = 'address_match';
        }
      }
    }

    // Strategy 3: Fuzzy client name match
    if (!score && searchClient) {
      const clientSim = similarityRatio(searchClient, clientFromName);
      if (clientSim > 0.8) {
        score = clientSim * 0.9;  // Cap at 0.72
        matchStrategy = 'fuzzy_client';
      }
    }

    // Strategy 4: Reference contains job identifier
    if (!score && searchRef) {
      const refSim = similarityRatio(searchRef, jobName);
      if (refSim > 0.6 || searchRef.includes(clientFromName)) {
        score = 0.70;
        matchStrategy = 'reference_match';
      }
    }

    if (score > 0) {
      candidates.push({ job, score, matchStrategy });
    }
  }

  // Sort by score, return best match
  candidates.sort((a, b) => b.score - a.score);

  if (candidates.length > 0) {
    const best = candidates[0];
    return {
      job: best.job,
      confidence: best.score,
      matchStrategy: best.matchStrategy,
      alternates: candidates.slice(1, 3)  // Include top 2 alternates
    };
  }

  return null;
}
```

### Task 3: Enhanced findOrCreateVendor()

Improve vendor matching with Levenshtein:

```javascript
async function findOrCreateVendor(vendorData, autoCreate = true) {
  if (!vendorData?.companyName) {
    return { vendor: null, isNew: false, confidence: 0 };
  }

  const vendorName = vendorData.companyName.trim();
  const normalizedName = vendorName.toLowerCase();

  // Load all vendors
  const { data: vendors, error } = await supabase
    .from('v2_vendors')
    .select('*');

  if (!error && vendors?.length > 0) {
    // Score all vendors
    const scored = vendors.map(v => {
      const vName = (v.name || '').toLowerCase();

      // Exact match
      if (vName === normalizedName) {
        return { vendor: v, score: 0.99, reason: 'exact' };
      }

      // Similarity match
      const sim = similarityRatio(normalizedName, vName);
      if (sim > 0.85) {
        return { vendor: v, score: sim, reason: 'fuzzy' };
      }

      // First word match (company name often starts with key identifier)
      const firstWord = normalizedName.split(/\s+/)[0];
      const vFirstWord = vName.split(/\s+/)[0];
      if (firstWord.length > 3 && vFirstWord.length > 3 &&
          similarityRatio(firstWord, vFirstWord) > 0.9) {
        return { vendor: v, score: 0.80, reason: 'first_word' };
      }

      return null;
    }).filter(Boolean);

    scored.sort((a, b) => b.score - a.score);

    if (scored.length > 0 && scored[0].score > 0.75) {
      return {
        vendor: scored[0].vendor,
        isNew: false,
        confidence: scored[0].score,
        matchReason: scored[0].reason,
        alternates: scored.slice(1, 3)
      };
    }
  }

  // No match found - create if allowed
  if (!autoCreate) {
    return { vendor: null, isNew: false, confidence: 0 };
  }

  // Create new vendor using standards
  const newVendor = {
    name: standards.toTitleCase(vendorName),
    phone: vendorData.phone ? standards.normalizePhone(vendorData.phone) : null,
    email: vendorData.email?.toLowerCase() || null,
    address: vendorData.address || null,
    trade: vendorData.tradeType || 'other'
  };

  const { data: created, error: createError } = await supabase
    .from('v2_vendors')
    .insert(newVendor)
    .select()
    .single();

  if (createError) {
    console.error('[AI-PO] Failed to create vendor:', createError.message);
    return { vendor: null, isNew: false, confidence: 0 };
  }

  return { vendor: created, isNew: true, confidence: 0.95 };
}
```
