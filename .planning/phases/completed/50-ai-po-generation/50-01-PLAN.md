### Task 1: Add PO Document Upload Endpoint

Create POST `/api/purchase-orders/process-document` endpoint:

```javascript
// In server/routes/purchase-orders.js

const { processPODocument } = require('../ai-po-processor');
const multer = require('multer');
const upload = multer({ storage: multer.memoryStorage() });

router.post('/process-document', upload.single('file'), asyncHandler(async (req, res) => {
  if (!req.file) {
    return res.status(400).json({ error: 'No file uploaded' });
  }

  const results = await processPODocument(
    req.file.buffer,
    req.file.originalname,
    req.file.mimetype
  );

  // Return structured response for frontend review
  res.json({
    success: results.success,
    extracted: results.extracted,
    matchedJob: results.matchedJob,
    vendor: results.vendor,
    lineItems: results.lineItems,
    po: results.po,
    confidence: results.confidence,
    messages: results.messages,
    needsReview: !results.po  // True if auto-create failed
  });
}));
```

### Task 2: Add Preview Mode (No Auto-Create)

Add `preview=true` query param to extract data without creating PO:

```javascript
router.post('/process-document', upload.single('file'), asyncHandler(async (req, res) => {
  const previewOnly = req.query.preview === 'true';

  // If preview, return extracted data without creating PO
  if (previewOnly) {
    const results = await processPODocument(req.file.buffer, req.file.originalname, req.file.mimetype);
    // Don't include auto-created PO in preview mode
    return res.json({
      ...results,
      po: null,
      needsReview: true
    });
  }

  // Normal flow with auto-create

}));
```
