<objective>
Add trades and dependencies management UI to the catalog page. Staff can view compatible trades (with labor rates) and dependency relationships in the product detail modal, and manage them in the product form modal.

Purpose: Enable staff to link catalog items to trades for accurate labor costing, and define installation dependencies for scheduling.

Output: Enhanced catalog UI with trades and dependencies management.
</objective>

<context>
@public/catalog.html - Existing modals
@public/js/catalog.js - Existing functions
@server/routes/selections.js - API endpoints from 70-02 (GET/POST/DELETE trades, dependencies)

**API endpoints available:**
- GET /api/selections/trades - List all available trades
- GET /api/selections/catalog/:id/trades - Get linked trades
- POST /api/selections/catalog/:id/trades - Add trade link
- DELETE /api/selections/catalog/:id/trades/:linkId - Remove trade link
- GET /api/selections/catalog/:id/dependencies - Get dependencies
- POST /api/selections/catalog/:id/dependencies - Add dependency
- DELETE /api/selections/catalog/:id/dependencies/:depId - Remove dependency

**Trades data structure:**
```json
{
  "id": "link-uuid",
  "trade_id": "trade-uuid",
  "is_primary": true,
  "labor_hours_override": 2.5,
  "hourly_rate_override": 85.00,
  "trade": { "id": "", "name": "Plumbing", "code": "PLUM" }
}
```

**Dependencies data structure:**
```json
{
  "id": "dep-uuid",
  "dependency_type": "must_follow",
  "depends_on_item_id": "item-uuid",
  "depends_on_category_id": null,
  "gap_days": 1,
  "notes": "Must install after cabinets",
  "depends_on_item": { "id": "", "name": "Kitchen Cabinets" }
}
```

```javascript
// ============================================================
// TRADES MANAGEMENT
// ============================================================

let allTrades = [];
let linkedTrades = [];

/**
 * Load all available trades
 */
async function loadAllTrades() {
  try {
    const res = await fetch('/api/selections/trades');
    if (!res.ok) throw new Error('Failed to load trades');
    allTrades = await res.json();
    populateTradeDropdown();
  } catch (err) {
    console.error('Failed to load trades:', err);
  }
}

/**
 * Populate the trade dropdown with available trades
 */
function populateTradeDropdown() {
  const select = document.getElementById('addTradeSelect');
  if (!select) return;

  select.innerHTML = '<option value="">Select a trade to add...</option>';
  allTrades.forEach(t => {
    // Skip already linked trades
    if (linkedTrades.some(lt => lt.trade_id === t.id)) return;
    select.innerHTML += `<option value="${t.id}">${escapeHtml(t.name)} (${t.code})</option>`;
  });
}

/**
 * Load linked trades for current product
 */
async function loadLinkedTrades(productId) {
  if (!productId) {
    linkedTrades = [];
    renderLinkedTrades();
    return;
  }

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/trades`);
    if (!res.ok) throw new Error('Failed to load trades');
    linkedTrades = await res.json();
    renderLinkedTrades();
    populateTradeDropdown(); // Update dropdown to exclude linked
  } catch (err) {
    console.error('Failed to load linked trades:', err);
    linkedTrades = [];
    renderLinkedTrades();
  }
}

/**
 * Render the list of linked trades in the form
 */
function renderLinkedTrades() {
  const container = document.getElementById('linkedTradesList');
  if (!container) return;

  if (linkedTrades.length === 0) {
    container.innerHTML = '<p class="no-items">No trades linked yet. Add trades below.</p>';
    return;
  }

  container.innerHTML = linkedTrades.map(t => {
    const trade = t.trade || {};
    return `
      <div class="linked-item">
        <div class="linked-item-info">
          <span class="item-name">${escapeHtml(trade.name || 'Unknown')}</span>
          ${t.is_primary ? '<span class="badge badge-primary">Primary</span>' : ''}
          ${t.labor_hours_override ? `<span class="item-detail">${t.labor_hours_override} hrs</span>` : ''}
          ${t.hourly_rate_override ? `<span class="item-detail">$${t.hourly_rate_override}/hr</span>` : ''}
        </div>
        <button type="button" class="btn btn-sm btn-danger" onclick="removeLinkedTrade('${t.id}')">Remove</button>
      </div>
    `;
  }).join('');
}

/**
 * Add a trade to the current product
 */
async function addTradeToProduct() {
  const productId = document.getElementById('editProductId').value;
  if (!productId) {
    showToast('Please save the product first before adding trades', 'info');
    return;
  }

  const tradeId = document.getElementById('addTradeSelect').value;
  if (!tradeId) {
    showToast('Please select a trade', 'error');
    return;
  }

  const laborHours = document.getElementById('addTradeLaborHours').value;
  const hourlyRate = document.getElementById('addTradeHourlyRate').value;
  const isPrimary = document.getElementById('addTradeIsPrimary').checked;

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/trades`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        trade_id: tradeId,
        is_primary: isPrimary,
        labor_hours_override: laborHours ? parseFloat(laborHours) : null,
        hourly_rate_override: hourlyRate ? parseFloat(hourlyRate) : null
      })
    });

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.error || 'Failed to add trade');
    }

    showToast('Trade linked', 'success');

    // Reset form
    document.getElementById('addTradeSelect').value = '';
    document.getElementById('addTradeLaborHours').value = '';
    document.getElementById('addTradeHourlyRate').value = '';
    document.getElementById('addTradeIsPrimary').checked = false;
    document.getElementById('addTradeOptions').style.display = 'none';

    // Reload trades
    await loadLinkedTrades(productId);
  } catch (err) {
    console.error('Failed to add trade:', err);
    showToast(err.message, 'error');
  }
}

/**
 * Remove a linked trade
 */
async function removeLinkedTrade(linkId) {
  const productId = document.getElementById('editProductId').value;
  if (!productId || !linkId) return;

  if (!confirm('Remove this trade link?')) return;

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/trades/${linkId}`, {
      method: 'DELETE'
    });

    if (!res.ok) throw new Error('Failed to remove trade');

    showToast('Trade removed', 'success');
    await loadLinkedTrades(productId);
  } catch (err) {
    console.error('Failed to remove trade:', err);
    showToast('Failed to remove trade', 'error');
  }
}
```

```javascript
// ============================================================
// DEPENDENCIES MANAGEMENT
// ============================================================

let linkedDependencies = [];

/**
 * Load dependencies for current product
 */
async function loadLinkedDependencies(productId) {
  if (!productId) {
    linkedDependencies = [];
    renderLinkedDependencies();
    return;
  }

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/dependencies`);
    if (!res.ok) throw new Error('Failed to load dependencies');
    linkedDependencies = await res.json();
    renderLinkedDependencies();
  } catch (err) {
    console.error('Failed to load dependencies:', err);
    linkedDependencies = [];
    renderLinkedDependencies();
  }
}

/**
 * Render the list of dependencies in the form
 */
function renderLinkedDependencies() {
  const container = document.getElementById('linkedDependenciesList');
  if (!container) return;

  if (linkedDependencies.length === 0) {
    container.innerHTML = '<p class="no-items">No dependencies defined. Add dependencies below.</p>';
    return;
  }

  const typeLabels = {
    must_precede: 'Must be BEFORE',
    must_follow: 'Must be AFTER',
    incompatible: 'Incompatible with'
  };

  container.innerHTML = linkedDependencies.map(d => {
    const targetName = d.depends_on_item?.name || d.depends_on_category?.name || 'Unknown';
    const typeLabel = typeLabels[d.dependency_type] || d.dependency_type;

    return `
      <div class="linked-item dependency-item-form">
        <div class="linked-item-info">
          <span class="dep-type-label">${typeLabel}</span>
          <span class="item-name">${escapeHtml(targetName)}</span>
          ${d.gap_days > 0 ? `<span class="item-detail">(${d.gap_days} day gap)</span>` : ''}
          ${d.notes ? `<span class="item-detail dep-notes-inline">${escapeHtml(d.notes)}</span>` : ''}
        </div>
        <button type="button" class="btn btn-sm btn-danger" onclick="removeLinkedDependency('${d.id}')">Remove</button>
      </div>
    `;
  }).join('');
}

/**
 * Add a dependency to the current product
 */
async function addDependencyToProduct() {
  const productId = document.getElementById('editProductId').value;
  if (!productId) {
    showToast('Please save the product first before adding dependencies', 'info');
    return;
  }

  const depType = document.getElementById('addDepType').value;
  if (!depType) {
    showToast('Please select a dependency type', 'error');
    return;
  }

  const targetType = document.getElementById('addDepTargetType').value;
  const categoryId = document.getElementById('addDepCategory').value;
  const itemId = document.getElementById('addDepItem').value;
  const gapDays = parseInt(document.getElementById('addDepGapDays').value) || 0;
  const notes = document.getElementById('addDepNotes').value.trim();

  // Validate target
  if (targetType === 'category' && !categoryId) {
    showToast('Please select a category', 'error');
    return;
  }
  if (targetType === 'item' && !itemId) {
    showToast('Please select an item', 'error');
    return;
  }

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/dependencies`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        dependency_type: depType,
        depends_on_category_id: targetType === 'category' ? categoryId : null,
        depends_on_item_id: targetType === 'item' ? itemId : null,
        gap_days: gapDays,
        notes: notes || null
      })
    });

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.error || 'Failed to add dependency');
    }

    showToast('Dependency added', 'success');

    // Reset form
    document.getElementById('addDepType').value = '';
    document.getElementById('addDepTarget').style.display = 'none';
    document.getElementById('addDepGapDays').value = '0';
    document.getElementById('addDepNotes').value = '';

    // Reload dependencies
    await loadLinkedDependencies(productId);
  } catch (err) {
    console.error('Failed to add dependency:', err);
    showToast(err.message, 'error');
  }
}

/**
 * Remove a linked dependency
 */
async function removeLinkedDependency(depId) {
  const productId = document.getElementById('editProductId').value;
  if (!productId || !depId) return;

  if (!confirm('Remove this dependency?')) return;

  try {
    const res = await fetch(`/api/selections/catalog/${productId}/dependencies/${depId}`, {
      method: 'DELETE'
    });

    if (!res.ok) throw new Error('Failed to remove dependency');

    showToast('Dependency removed', 'success');
    await loadLinkedDependencies(productId);
  } catch (err) {
    console.error('Failed to remove dependency:', err);
    showToast('Failed to remove dependency', 'error');
  }
}

/**
 * Populate dependency category dropdown
 */
function populateDepCategoryDropdown() {
  const select = document.getElementById('addDepCategory');
  if (!select) return;

  select.innerHTML = '<option value="">Select category...</option>';
  allCategories.forEach(cat => {
    select.innerHTML += `<option value="${cat.id}">${escapeHtml(cat.name)}</option>`;
    if (cat.children) {
      cat.children.forEach(child => {
        select.innerHTML += `<option value="${child.id}">&nbsp;&nbsp;${escapeHtml(child.name)}</option>`;
      });
    }
  });
}

/**
 * Load catalog items for dependency item dropdown
 */
async function loadDepItemDropdown(categoryId) {
  const select = document.getElementById('addDepItem');
  if (!select) return;

  select.innerHTML = '<option value="">Loading...</option>';

  try {
    let url = '/api/selections/catalog';
    if (categoryId) url += `?category_id=${categoryId}`;

    const res = await fetch(url);
    if (!res.ok) throw new Error('Failed to load items');
    const items = await res.json();

    select.innerHTML = '<option value="">Select item...</option>';
    items.forEach(item => {
      // Don't allow self-dependency
      const currentId = document.getElementById('editProductId').value;
      if (item.id === currentId) return;
      select.innerHTML += `<option value="${item.id}">${escapeHtml(item.name)}</option>`;
    });
  } catch (err) {
    console.error('Failed to load items:', err);
    select.innerHTML = '<option value="">Error loading items</option>';
  }
}
```
