## Verification Script

```javascript
const BASE = 'http://localhost:3001/api/order-optimizer';

async function verify() {
  // 1. Parse material list
  const parsed = await fetch(`${BASE}/parse-list`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ text: '50 2x4x8 studs\n20 sheets drywall' })
  }).then(r => r.json());
  console.log('Parsed:', parsed.parsed_count, 'items');

  // 2. Get waste factors
  const factors = await fetch(`${BASE}/waste-factors`).then(r => r.json());
  console.log('Waste factors:', factors.length);

  // 3. Run optimization
  const result = await fetch(`${BASE}/optimize`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      items: [{ description: '2x4x8 stud grade', quantity: 50, unit: 'ea' }],
      include_waste: true
    })
  }).then(r => r.json());
  console.log('Optimization:', result.summary.matched_items, 'matched,', result.summary.vendor_count, 'vendors');
  console.log('Savings:', '$' + result.summary.savings_amount?.toFixed(2));

  // 4. List orders
  const orders = await fetch(`${BASE}/orders`).then(r => r.json());
  console.log('Saved orders:', orders.length);

  console.log('\n✓ All endpoints responding');
}

verify().catch(console.error);
```
