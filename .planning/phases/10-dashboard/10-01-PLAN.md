async function loadAlerts() {
  const container = document.getElementById('alertCards');
  try {
    const res = await fetch('/api/dashboard/alerts');
    const alerts = await res.json();

    const cards = [];

    // Upcoming inspections
    if (alerts.inspections?.count > 0) {
      cards.push({
        icon: '🔍',
        count: alerts.inspections.count,
        label: 'Upcoming Inspections',
        type: 'info',
        href: 'inspections.html?filter=scheduled'
      });
    }

    // Budget overruns
    if (alerts.budgetOverruns?.count > 0) {
      cards.push({
        icon: '💰',
        count: alerts.budgetOverruns.count,
        label: 'Budget Overruns',
        type: 'danger',
        href: 'budgets.html'
      });
    }

    // Pending approvals
    const pendingTotal = (alerts.pendingApprovals?.invoices || 0) + (alerts.pendingApprovals?.pos || 0);
    if (pendingTotal > 0) {
      cards.push({
        icon: '✅',
        count: pendingTotal,
        label: 'Pending Approvals',
        type: 'warning',
        href: 'index.html?status=needs_approval'
      });
    }

    // Open punch items
    if (alerts.punchItems?.count > 0) {
      cards.push({
        icon: '📝',
        count: alerts.punchItems.count,
        label: 'Open Punch Items',
        type: 'warning',
        href: 'punch-lists.html'
      });
    }

    // Overdue inspections
    if (alerts.overdueInspections?.count > 0) {
      cards.push({
        icon: '⚠️',
        count: alerts.overdueInspections.count,
        label: 'Overdue Inspections',
        type: 'danger',
        href: 'inspections.html?filter=overdue'
      });
    }

    if (cards.length === 0) {
      container.innerHTML = '<div class="empty-state"><p>No items need attention - you\'re all caught up!</p></div>';
      return;
    }

    container.innerHTML = cards.map(card => `
      <a href="${card.href}" class="alert-card ${card.type}">
        <div class="alert-card-icon">${card.icon}</div>
        <div class="alert-card-content">
          <div class="alert-card-count">${card.count}</div>
          <div class="alert-card-label">${card.label}</div>
        </div>
      </a>
    `).join('');
  } catch (err) {
    console.error('Failed to load alerts:', err);
    container.innerHTML = '<div class="empty-state"><p>Failed to load alerts</p></div>';
  }
}
```

// Get recent activity across all entities
router.get('/activity', async (req, res) => {
  try {
    const limit = parseInt(req.query.limit) || 10;
    const activities = [];

    // Recent invoices
    const { data: invoices } = await supabase
      .from('v2_invoices')
      .select('id, invoice_number, amount, status, created_at, v2_vendors(name)')
      .order('created_at', { ascending: false })
      .limit(5);

    invoices?.forEach(inv => {
      activities.push({
        type: 'invoice',
        icon: '📄',
        text: `Invoice #${inv.invoice_number || 'N/A'} from ${inv.v2_vendors?.name || 'Unknown'} - ${formatMoney(inv.amount)}`,
        status: inv.status,
        timestamp: inv.created_at
      });
    });

    // Recent PO activity
    const { data: poActivity } = await supabase
      .from('v2_po_activity')
      .select('id, action, details, created_at, v2_purchase_orders(po_number)')
      .order('created_at', { ascending: false })
      .limit(5);

    poActivity?.forEach(act => {
      activities.push({
        type: 'po',
        icon: '📋',
        text: `PO ${act.v2_purchase_orders?.po_number}: ${act.action}`,
        timestamp: act.created_at
      });
    });

    // Recent inspections
    const { data: inspections } = await supabase
      .from('v2_inspections')
      .select('id, type, status, result_date, v2_jobs(name)')
      .not('result_date', 'is', null)
      .order('result_date', { ascending: false })
      .limit(5);

    inspections?.forEach(insp => {
      activities.push({
        type: 'inspection',
        icon: '🔍',
        text: `${insp.type} inspection ${insp.status} - ${insp.v2_jobs?.name}`,
        status: insp.status,
        timestamp: insp.result_date
      });
    });

    // Sort by timestamp and limit
    activities.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));

    res.json(activities.slice(0, limit));
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

function formatMoney(amount) {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 0
  }).format(amount || 0);
}
```

Update `loadActivity()` to use the new endpoint:

async function loadActivity() {
  const container = document.getElementById('activityFeed');
  try {
    const res = await fetch('/api/dashboard/activity?limit=10');
    const activities = await res.json();

    if (!activities || activities.length === 0) {
      container.innerHTML = '<div class="empty-state"><p>No recent activity</p></div>';
      return;
    }

    container.innerHTML = activities.map(act => `
      <div class="activity-item">
        <div class="activity-dot" style="background: ${getActivityColor(act.type)}"></div>
        <div class="activity-content">
          <div class="activity-text">
            <span class="activity-icon">${act.icon}</span>
            ${act.text}
          </div>
          <div class="activity-time">${formatDate(act.timestamp)}</div>
        </div>
      </div>
    `).join('');
  } catch (err) {
    console.error('Failed to load activity:', err);
    container.innerHTML = '<div class="empty-state"><p>Failed to load activity</p></div>';
  }
}

function getActivityColor(type) {
  const colors = {
    invoice: 'var(--accent-blue)',
    po: 'var(--accent-green)',
    inspection: '#a855f7',
    punch: 'var(--accent-orange)'
  };
  return colors[type] || 'var(--accent-blue)';
}
